const FRAGMENT_SHADER_SOURCE: &str = r#"#version 300 es
precision highp float;
in vec3 v_normal;
in vec3 v_frag_pos;
in vec2 v_uv;
in float v_face_id;
uniform vec3 u_light_dir;
uniform vec3 u_base_color;
uniform vec3 u_edge_color;
out vec4 frag_color;

const float PIP_RADIUS = 0.07;
const vec3 PIP_COLOR = vec3(0.02, 0.02, 0.02);

bool in_pip(vec2 uv, vec2 center) {
    return distance(uv, center) < PIP_RADIUS;
}

bool is_pip_fragment(float face_id, vec2 uv) {
    int face = int(face_id + 0.5);
    if (face == 1) {
        return in_pip(uv, vec2(0.5, 0.5));
    } else if (face == 2) {
        return in_pip(uv, vec2(0.25, 0.25)) || in_pip(uv, vec2(0.75, 0.75));
    } else if (face == 3) {
        return in_pip(uv, vec2(0.25, 0.25)) || in_pip(uv, vec2(0.5, 0.5)) || in_pip(uv, vec2(0.75, 0.75));
    } else if (face == 4) {
        return in_pip(uv, vec2(0.25, 0.25)) || in_pip(uv, vec2(0.75, 0.25))
            || in_pip(uv, vec2(0.25, 0.75)) || in_pip(uv, vec2(0.75, 0.75));
    } else if (face == 5) {
        return in_pip(uv, vec2(0.25, 0.25)) || in_pip(uv, vec2(0.75, 0.25))
            || in_pip(uv, vec2(0.5, 0.5))
            || in_pip(uv, vec2(0.25, 0.75)) || in_pip(uv, vec2(0.75, 0.75));
    } else if (face == 6) {
        return in_pip(uv, vec2(0.25, 0.2)) || in_pip(uv, vec2(0.75, 0.2))
            || in_pip(uv, vec2(0.25, 0.5)) || in_pip(uv, vec2(0.75, 0.5))
            || in_pip(uv, vec2(0.25, 0.8)) || in_pip(uv, vec2(0.75, 0.8));
    }
    return false;
}

void main() {
    vec3 normal = normalize(v_normal);
    float diff = max(dot(normal, normalize(u_light_dir)), 0.0);
    vec3 base = is_pip_fragment(v_face_id, v_uv) ? PIP_COLOR : u_base_color;

    // Render face edges with a distinct color to make the cube shape visible.
    float edge_dist = min(min(v_uv.x, v_uv.y), min(1.0 - v_uv.x, 1.0 - v_uv.y));
    if (edge_dist < 0.08 && !is_pip_fragment(v_face_id, v_uv)) {
        base = u_edge_color;
    }

    vec3 ambient = base * 0.35;
    vec3 diffuse = base * diff * 0.65;
    frag_color = vec4(ambient + diffuse, 1.0);
}
"#;
