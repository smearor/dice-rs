1use std::rc::Rc;
2
3use gtk4::prelude::*;
4
5use crate::config::app_settings::AppSettings;
6use crate::config::app_settings::AppSettingsData;

8/// Settings dialog with toggle switches for UI display options.
9pub struct SettingsDialog {
10    dialog: gtk4::Window,
11}
12
13impl SettingsDialog {
14    /// Create a new settings dialog.
15    pub fn new(parent: &gtk4::ApplicationWindow, settings: &AppSettings) -> Self {
16        let dialog = gtk4::Window::builder()
17            .title("Settings")
18            .modal(true)
19            .transient_for(parent)
20            .default_width(400)
21            .default_height(400)
22            .build();
23
24        let content = gtk4::Box::builder()
25            .orientation(gtk4::Orientation::Vertical)
26            .spacing(12)
27            .margin_start(24)
28            .margin_end(24)
29            .margin_top(24)
30            .margin_bottom(24)
31            .build();
32
33        let current = settings.get();
34
35        let show_dice_3d_switch = gtk4::Switch::builder().active(current.show_dice_3d).build();
36        content.append(&Self::create_setting_row("Show 3D dice", &show_dice_3d_switch));
37
38        let rotate_dice_3d_switch = gtk4::Switch::builder().active(current.rotate_dice_3d).build();
39        content.append(&Self::create_setting_row("Rotate 3D dice", &rotate_dice_3d_switch));
40
41        let show_stability_switch = gtk4::Switch::builder().active(current.show_stability_indicator).build();
42        content.append(&Self::create_setting_row("Show stability indicator", &show_stability_switch));
43
44        let show_tap_switch = gtk4::Switch::builder().active(current.show_tap_controls).build();
45        content.append(&Self::create_setting_row("Show tap indicator and controls", &show_tap_switch));
46
47        let show_led_switch = gtk4::Switch::builder().active(current.show_led_controls).build();
48        content.append(&Self::create_setting_row("Show LED controls", &show_led_switch));
49
50        let show_battery_switch = gtk4::Switch::builder().active(current.show_battery_indicator).build();
51        content.append(&Self::create_setting_row("Show battery indicator", &show_battery_switch));
52
53        let show_dice_type_switch = gtk4::Switch::builder().active(current.show_dice_type_selector).build();
54        content.append(&Self::create_setting_row("Show dice type selector", &show_dice_type_switch));
55
56        let show_history_switch = gtk4::Switch::builder().active(current.show_roll_history).build();
57        content.append(&Self::create_setting_row("Show roll history", &show_history_switch));
58
59        let settings_clone = settings.clone();
60        let s1 = show_dice_3d_switch.clone();
61        let s2 = rotate_dice_3d_switch.clone();
62        let s3 = show_stability_switch.clone();
63        let s4 = show_tap_switch.clone();
64        let s5 = show_led_switch.clone();
65        let s6 = show_battery_switch.clone();
66        let s7 = show_dice_type_switch.clone();
67        let s8 = show_history_switch.clone();
68
69        let emit_settings = Rc::new(move || {
70            let mut data = AppSettingsData {
71                show_dice_3d: s1.is_active(),
72                rotate_dice_3d: s2.is_active(),
73                show_stability_indicator: s3.is_active(),
74                show_tap_controls: s4.is_active(),
75                show_led_controls: s5.is_active(),
76                show_battery_indicator: s6.is_active(),
77                show_dice_type_selector: s7.is_active(),
78                show_roll_history: s8.is_active(),
79                compact_mode: false,
80            };
81            data.compact_mode = settings_clone.get().compact_mode;
82            settings_clone.set(data);
83        });
84
85        let e = emit_settings.clone();
86        show_dice_3d_switch.connect_notify_local(Some("active"), move |_, _| {
87            e();
88        });
89        let e = emit_settings.clone();
90        rotate_dice_3d_switch.connect_notify_local(Some("active"), move |_, _| {
91            e();
92        });
93        let e = emit_settings.clone();
94        show_stability_switch.connect_notify_local(Some("active"), move |_, _| {
95            e();
96        });
97        let e = emit_settings.clone();
98        show_tap_switch.connect_notify_local(Some("active"), move |_, _| {
99            e();
100        });
101        let e = emit_settings.clone();
102        show_led_switch.connect_notify_local(Some("active"), move |_, _| {
103            e();
104        });
105        let e = emit_settings.clone();
106        show_battery_switch.connect_notify_local(Some("active"), move |_, _| {
107            e();
108        });
109        let e = emit_settings.clone();
110        show_dice_type_switch.connect_notify_local(Some("active"), move |_, _| {
111            e();
112        });
113        let e = emit_settings.clone();
114        show_history_switch.connect_notify_local(Some("active"), move |_, _| {
115            e();
116        });
117
118        dialog.set_child(Some(&content));
119
120        Self { dialog }
121    }
122
123    /// Create a single setting row with a label and switch.
124    fn create_setting_row(label: &str, switch: &gtk4::Switch) -> gtk4::Box {
125        let row = gtk4::Box::builder().orientation(gtk4::Orientation::Horizontal).spacing(12).build();
126        row.append(&gtk4::Label::builder().label(label).hexpand(true).halign(gtk4::Align::Start).build());
127        switch.set_halign(gtk4::Align::End);
128        row.append(switch);
129        row
130    }
131
132    /// Present the dialog.
133    pub fn present(&self) {
134        self.dialog.present();
135    }
136}
