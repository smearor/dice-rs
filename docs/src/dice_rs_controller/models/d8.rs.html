1use glam::Vec3;
2
3use crate::models::DiceModel;
4use crate::models::DiceModelTrait;
5use crate::models::add_tri_face;

7/// D8 octahedron model with 8 triangular faces.
8///
9/// Vertices on the coordinate axes. Face values 1-8, opposite faces sum to 9.
10pub struct D8Model;
11
12impl DiceModelTrait for D8Model {
13    fn model(&self) -> DiceModel {
14        let v = [
15            [1.0, 0.0, 0.0],  // 0: +X
16            [-1.0, 0.0, 0.0], // 1: -X
17            [0.0, 1.0, 0.0],  // 2: +Y
18            [0.0, -1.0, 0.0], // 3: -Y
19            [0.0, 0.0, 1.0],  // 4: +Z
20            [0.0, 0.0, -1.0], // 5: -Z
21        ];
22
23        // Faces: (i0, i1, i2, face_id) — 8 triangles, outward-facing
24        // Opposite pairs: 1↔8, 2↔7, 3↔6, 4↔5
25        let faces: [(usize, usize, usize, f32); 8] = [
26            (2, 4, 0, 1.0), // +X +Y +Z
27            (2, 0, 5, 2.0), // +X +Y -Z
28            (2, 1, 4, 3.0), // -X +Y +Z
29            (2, 5, 1, 4.0), // -X +Y -Z
30            (3, 0, 4, 5.0), // +X -Y +Z
31            (3, 5, 0, 6.0), // +X -Y -Z
32            (3, 4, 1, 7.0), // -X -Y +Z
33            (3, 1, 5, 8.0), // -X -Y -Z
34        ];
35
36        let mut positions = Vec::with_capacity(24 * 3);
37        let mut normals = Vec::with_capacity(24 * 3);
38        let mut uvs = Vec::with_capacity(24 * 2);
39        let mut face_ids = Vec::with_capacity(24);
40        let mut indices = Vec::with_capacity(24);
41
42        for (i0, i1, i2, fid) in faces {
43            let e01 = Vec3::from(v[i1]) - Vec3::from(v[i0]);
44            let e02 = Vec3::from(v[i2]) - Vec3::from(v[i0]);
45            let n: [f32; 3] = e01.cross(e02).normalize().into();
46            add_tri_face(&mut positions, &mut normals, &mut uvs, &mut face_ids, &mut indices, v[i0], v[i1], v[i2], n, fid);
47        }
48
49        DiceModel {
50            positions,
51            normals,
52            uvs,
53            face_ids,
54            indices,
55            face_shape: 0,
56            is_d10x: false,
57        }
58    }
59}
