1/// Vertex data for a 3D dice model.
2///
3/// Contains all per-vertex attributes needed for OpenGL rendering:
4/// positions, normals, texture coordinates, face identifiers, and
5/// an index buffer. Also carries metadata used by the fragment shader
6/// to select the correct edge detection and number rendering logic.
7pub struct DiceModel {
8    /// Flat array of vertex positions, 3 floats per vertex (x, y, z).
9    pub positions: Vec<f32>,
10    /// Flat array of vertex normals, 3 floats per vertex (nx, ny, nz).
11    /// All vertices of a face share the same normal for flat shading.
12    pub normals: Vec<f32>,
13    /// Flat array of texture coordinates, 2 floats per vertex (u, v).
14    /// UV layout depends on the face shape — triangle, quad, kite, or pentagon.
15    pub uvs: Vec<f32>,
16    /// Flat array of face identifiers, 1 float per vertex.
17    /// Each vertex carries the face number it belongs to (1-indexed).
18    /// Used by the shader to render the correct pip pattern or digit.
19    pub face_ids: Vec<f32>,
20    /// Index buffer for indexed drawing, referencing vertices by position.
21    /// Each group of 3 indices forms one triangle.
22    pub indices: Vec<u32>,
23    /// Controls shader edge detection and number positioning.
24    /// 0 = triangle, 1 = quad, 2 = pentagon, 3 = kite.
25    pub face_shape: i32,
26    /// Whether this die is a D10X (tens die showing 00, 10, 20, ..., 90).
27    /// When true, the shader renders two-digit values instead of single digits.
28    pub is_d10x: bool,
29}
