6/// UI update commands sent from the tokio event loop to the GTK main thread.
7///
8/// `EventController` produces these variants on a background tokio task and
9/// forwards them via a `std::sync::mpsc::Sender`. The GTK side (in `DiceRow`)
10/// polls the corresponding receiver with `glib::timeout_add_local` and applies
11/// each update to the appropriate widgets. This keeps all widget mutation on
12/// the GTK main thread while dice I/O runs off-thread.
13pub enum UiUpdate {
14    /// The dice started rolling — reset face display to the rolling state.
15    Rolling,
16    /// The dice settled on a face with full stability.
17    Stable {
18        /// The resolved face value after the roll.
19        face: FaceValue,
20        /// The acceleration vector at rest, used to orient the 3D model.
21        acceleration: Acceleration,
22    },
23    /// The dice settled but is tilted (not flat on a face).
24    TiltStable {
25        /// The resolved face value despite the tilt.
26        face: FaceValue,
27        /// The acceleration vector at rest, used to orient the 3D model.
28        acceleration: Acceleration,
29    },
30    /// The dice reported a stable face but the reading is likely spurious.
31    FakeStable {
32        /// The suspected face value.
33        face: FaceValue,
34        /// The acceleration vector at rest, used to orient the 3D model.
35        acceleration: Acceleration,
36    },
37    /// The dice settled after being moved (not thrown).
38    MoveStable {
39        /// The resolved face value after the move.
40        face: FaceValue,
41        /// The acceleration vector at rest, used to orient the 3D model.
42        acceleration: Acceleration,
43    },
44    /// The dice charging state changed.
45    Charging {
46        /// The new charging state (charging or not charging).
47        state: ChargingState,
48    },
49    /// A single tap was detected on the dice.
50    Tap,
51    /// A double tap was detected on the dice.
52    DoubleTap,
53    /// The BLE connection to the dice was lost.
54    Disconnected,
55    /// A periodic battery level reading completed.
56    BatteryLevel(
57        /// The most recently reported battery level.
58        BatteryLevel,
59    ),
60}
