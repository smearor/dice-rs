16/// Interval for periodic battery level refresh.
17const BATTERY_REFRESH_INTERVAL_SECS: u64 = 30;
18
19/// Interval for battery level refresh while charging.
20const BATTERY_CHARGING_REFRESH_INTERVAL_SECS: u64 = 1;
21
22/// Delay before first reconnect attempt after a disconnect.
23const RECONNECT_INITIAL_DELAY_SECS: u64 = 2;
24
25/// Bridges async dice events into a channel for the UI to consume.
26///
27/// The event loop runs on a `tokio::spawn` task so it never blocks the GTK
28/// main thread. UI updates are marshaled back via a `std::sync::mpsc` channel
29/// that the caller polls on the GTK side.
30pub struct EventController {
31    dice: Dice,
32    manager: Arc<DiceManager>,
33}
34
35impl EventController {
36    /// Create a new event controller.
37    pub fn new(dice: Dice, manager: Arc<DiceManager>) -> Self {
38        Self { dice, manager }
39    }
40
41    /// Start listening for dice events and sending UI updates via `sender`.
42    pub fn start(&self, sender: mpsc::Sender<UiUpdate>) {
43        // Tokio task: event loop - runs entirely off the GTK main thread.
44        let dice = self.dice.clone();
45        let dice_name = dice.name().to_string();
46        let manager = self.manager.clone();
47        let event_sender = sender.clone();
48        let charging_flag = Arc::new(AtomicBool::new(false));
49        let charging_flag_for_battery = charging_flag.clone();
50        let charging_notify = Arc::new(Notify::new());
51        let charging_notify_for_battery = charging_notify.clone();
52        tokio::spawn(async move {
53            let mut receiver = dice.subscribe();
54
55            loop {
56                match receiver.recv().await {
57                    Ok(DiceEvent::RollStart) => {
58                        if !charging_flag.load(Ordering::Relaxed) {
59                            let _ = event_sender.send(UiUpdate::Rolling);
60                        }
61                    }
62                    Ok(DiceEvent::Stable { face, acceleration }) => {
63                        if !charging_flag.load(Ordering::Relaxed) {
64                            let _ = event_sender.send(UiUpdate::Stable { face, acceleration });
65                        }
66                    }
67                    Ok(DiceEvent::TiltStable { face, acceleration }) => {
68                        if !charging_flag.load(Ordering::Relaxed) {
69                            let _ = event_sender.send(UiUpdate::TiltStable { face, acceleration });
70                        }
71                    }
72                    Ok(DiceEvent::FakeStable { face, acceleration }) => {
73                        if !charging_flag.load(Ordering::Relaxed) {
74                            let _ = event_sender.send(UiUpdate::FakeStable { face, acceleration });
75                        }
76                    }
77                    Ok(DiceEvent::MoveStable { face, acceleration }) => {
78                        if !charging_flag.load(Ordering::Relaxed) {
79                            let _ = event_sender.send(UiUpdate::MoveStable { face, acceleration });
80                        }
81                    }
82                    Ok(DiceEvent::Charging { state }) => {
83                        let charging = bool::from(state);
84                        charging_flag.store(charging, Ordering::Relaxed);
85                        if charging {
86                            charging_notify.notify_one();
87                        }
88                        let _ = event_sender.send(UiUpdate::Charging { state });
89                    }
90                    Ok(DiceEvent::Tap) => {
91                        let _ = event_sender.send(UiUpdate::Tap);
92                    }
93                    Ok(DiceEvent::DoubleTap) => {
94                        let _ = event_sender.send(UiUpdate::DoubleTap);
95                    }
96                    Ok(DiceEvent::Disconnected) => {
97                        let _ = event_sender.send(UiUpdate::Disconnected);
98
99                        // Auto-reconnect on the tokio threadpool.
100                        tokio::time::sleep(Duration::from_secs(RECONNECT_INITIAL_DELAY_SECS)).await;
101                        match manager.reconnect(&dice).await {
102                            Ok(()) => {
103                                debug!(device = %dice_name, "reconnected successfully, resuming event loop");
104                                receiver = dice.subscribe();
105                            }
106                            Err(error) => {
107                                debug!(device = %dice_name, error = %error, "reconnect failed permanently");
108                                break;
109                            }
110                        }
111                    }
112                    Err(RecvError::Lagged(skipped)) => {
113                        debug!(device = %dice_name, skipped, "event receiver lagged, continuing");
114                    }
115                    Err(RecvError::Closed) => {
116                        debug!(device = %dice_name, "event channel closed, stopping event controller");
117                        break;
118                    }
119                }
120            }
121        });
122
123        // Tokio task: periodic battery refresh - fully off the GTK main thread.
124        // Polls more frequently while the dice is charging.
125        let battery_dice = self.dice.clone();
126        let battery_name = self.dice.name().to_string();
127        let battery_sender = sender;
128        tokio::spawn(async move {
129            tokio::time::sleep(Duration::from_secs(2)).await;
130            match battery_dice.get_battery_level().await {
131                Ok(level) => {
132                    let _ = battery_sender.send(UiUpdate::BatteryLevel(level));
133                }
134                Err(error) => debug!(device = %battery_name, error = %error, "initial battery fetch failed"),
135            }
136
137            loop {
138                // Use dice.charging_state() as ground truth - the broadcast event
139                // can be missed (lag), but the AtomicU8 in DiceInner is always
140                // updated by the notification task.
141                let state = battery_dice.charging_state();
142                let charging = bool::from(state);
143                let prev = charging_flag_for_battery.swap(charging, Ordering::Relaxed);
144                if charging != prev {
145                    debug!(device = %battery_name, charging, prev, "charging state sync via battery task");
146                    let _ = battery_sender.send(UiUpdate::Charging { state });
147                }
148
149                let interval = if charging {
150                    BATTERY_CHARGING_REFRESH_INTERVAL_SECS
151                } else {
152                    BATTERY_REFRESH_INTERVAL_SECS
153                };
154                // When not charging, wake early if charging starts.
155                if charging {
156                    tokio::time::sleep(Duration::from_secs(interval)).await;
157                } else {
158                    tokio::select! {
159                        _ = tokio::time::sleep(Duration::from_secs(interval)) => {}
160                        _ = charging_notify_for_battery.notified() => {}
161                    }
162                }
163                match battery_dice.is_connected().await {
164                    Ok(true) => {}
165                    Ok(false) => break,
166                    Err(error) => {
167                        debug!(device = %battery_name, error = %error, "battery refresh: is_connected failed");
168                        continue;
169                    }
170                }
171                match battery_dice.get_battery_level().await {
172                    Ok(level) => {
173                        debug!(device = %battery_name, charging, level = level.get(), "battery refresh succeeded");
174                        let _ = battery_sender.send(UiUpdate::BatteryLevel(level));
175                    }
176                    Err(error) => debug!(device = %battery_name, error = %error, charging, "battery refresh failed"),
177                }
178            }
179        });
180    }
181}
