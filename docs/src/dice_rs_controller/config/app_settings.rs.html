9/// Application display settings data.
10///
11/// Controls visibility of various UI elements in the dice controller.
12#[derive(Clone, Debug, Serialize, Deserialize)]
13pub struct AppSettingsData {
14    /// Whether the 3D dice view is shown.
15    pub show_dice_3d: bool,
16    /// Whether the 3D dice rotates continuously.
17    pub rotate_dice_3d: bool,
18    /// Whether the stability indicator is shown.
19    pub show_stability_indicator: bool,
20    /// Whether the tap indicator and tap controls are shown.
21    pub show_tap_controls: bool,
22    /// Whether the LED controls are shown.
23    pub show_led_controls: bool,
24    /// Whether the battery indicator is shown.
25    pub show_battery_indicator: bool,
26    /// Whether the dice type selector is shown.
27    pub show_dice_type_selector: bool,
28    /// Whether the roll history is shown.
29    pub show_roll_history: bool,
30    /// Whether compact mode is enabled (single-line rows).
31    pub compact_mode: bool,
32}

50/// Type alias for the settings change listener callback.
51type SettingsListener = Box<dyn Fn(AppSettingsData) + 'static>;
52
53/// Shared, cloneable application settings with change notification.
54///
55/// All clones share the same underlying state. When settings are updated
56/// via [`AppSettings::set`], all registered listeners are invoked.
57#[derive(Clone)]
58pub struct AppSettings {
59    data: Rc<RefCell<AppSettingsData>>,
60    listeners: Rc<RefCell<Vec<SettingsListener>>>,
61}
62
63impl AppSettings {
64    /// Create a new settings instance, loading from disk if available.
65    pub fn new() -> Self {
66        let data = load_app_settings().unwrap_or_default();
67        Self {
68            data: Rc::new(RefCell::new(data)),
69            listeners: Rc::new(RefCell::new(Vec::new())),
70        }
71    }
72
73    /// Get a snapshot of the current settings.
74    pub fn get(&self) -> AppSettingsData {
75        self.data.borrow().clone()
76    }
77
78    /// Update settings and notify all listeners.
79    pub fn set(&self, data: AppSettingsData) {
80        *self.data.borrow_mut() = data.clone();
81        save_app_settings(&data);
82        for listener in self.listeners.borrow().iter() {
83            listener(data.clone());
84        }
85    }
86
87    /// Register a callback that is invoked whenever settings change.
88    pub fn connect_changed<F>(&self, callback: F)
89    where
90        F: Fn(AppSettingsData) + 'static,
91    {
92        self.listeners.borrow_mut().push(Box::new(callback));
93    }
94}
95
96impl Default for AppSettings {
97    fn default() -> Self {
98        Self::new()
99    }
