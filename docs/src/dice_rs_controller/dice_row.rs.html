18/// Map a DiceColor to a CSS class name for the border.
19fn dice_color_to_css_class(color: DiceColor) -> &'static str {
20    match color {
21        DiceColor::Black => "dice-border-black",
22        DiceColor::Red => "dice-border-red",
23        DiceColor::Green => "dice-border-green",
24        DiceColor::Blue => "dice-border-blue",
25        DiceColor::Yellow => "dice-border-yellow",
26        DiceColor::Orange => "dice-border-orange",
27    }
28}

30/// A list row representing a single connected dice.
31pub struct DiceRow {
32    container: gtk4::Box,
33}
34
35impl DiceRow {
36    /// Create a new dice row for a connected dice.
37    pub fn new(dice: Dice, manager: Arc<DiceManager>) -> Self {
38        let face_display = FaceDisplay::new();
39        let battery_indicator = BatteryIndicator::new();
40        let led_controls = LedControls::new();
41        let dice_3d = Dice3D::new();
42        let roll_history = RollHistory::new();
43        let tap_indicator = TapIndicator::new();
44        led_controls.set_dice(dice.clone());
45
46        // Left side: 3D dice view.
47        let dice_3d_frame = gtk4::Frame::builder()
48            .css_classes(vec!["dice-3d-frame"])
49            .width_request(120)
50            .height_request(120)
51            .child(dice_3d.widget())
52            .build();
53
54        // Right side: face value, stability, battery, history, LED controls.
55        let info_box = gtk4::Box::builder().orientation(gtk4::Orientation::Vertical).spacing(8).hexpand(true).build();
56
57        let header = gtk4::Box::builder().orientation(gtk4::Orientation::Horizontal).spacing(12).build();
58        header.append(face_display.widget());
59        header.append(face_display.stability_label());
60        header.append(tap_indicator.widget());
61
62        let battery_row = gtk4::Box::builder().orientation(gtk4::Orientation::Horizontal).spacing(8).build();
63        battery_row.append(battery_indicator.label());
64        let level_bar = battery_indicator.level_bar();
65        level_bar.set_hexpand(true);
66        level_bar.set_valign(gtk4::Align::Center);
67        level_bar.add_css_class("battery-level-bar");
68        battery_row.append(level_bar);
69
70        info_box.append(&header);
71        info_box.append(roll_history.widget());
72        info_box.append(led_controls.widget());
73        info_box.append(&battery_row);
74
75        let container = gtk4::Box::builder()
76            .orientation(gtk4::Orientation::Horizontal)
77            .spacing(12)
78            .margin_start(12)
79            .margin_end(12)
80            .margin_top(12)
81            .margin_bottom(12)
82            .css_classes(vec!["dice-row"])
83            .build();
84        container.append(&dice_3d_frame);
85        container.append(&info_box);
86
87        // Apply colored border based on dice physical color.
88        let container_clone = container.clone();
89        let dice_for_color = dice.clone();
90        let face_display_for_color = face_display.clone();
91        let roll_history_for_color = roll_history.clone();
92        let dice_3d_for_color = dice_3d.clone();
93        glib::spawn_future_local(async move {
94            if let Ok(color) = dice_for_color.get_color().await {
95                container_clone.add_css_class(dice_color_to_css_class(color));
96                face_display_for_color.set_dice_color(color);
97                roll_history_for_color.set_dice_color(color);
98                dice_3d_for_color.set_color(color);
99            }
100        });
101
102        let controller = EventController::new(dice, manager, face_display, battery_indicator, dice_3d, roll_history, tap_indicator);
103        controller.start();
104
105        // --- Drag-and-drop reordering ---
106
107        // DragSource: provides the container's index within its parent Box.
108        let drag_source = gtk4::DragSource::builder().actions(gdk::DragAction::MOVE).build();
109
110        let drag_container = container.clone();
111        drag_source.connect_prepare(move |_source, _x, _y| {
112            let parent = drag_container.parent()?;
113            let parent = parent.downcast::<gtk4::Box>().ok()?;
114            let mut index = 0i32;
115            let mut child = parent.first_child();
116            while let Some(c) = child {
117                if c == drag_container {
118                    let value = glib::Value::from(index);
119                    return Some(gdk::ContentProvider::for_value(&value));
120                }
121                index += 1;
122                child = c.next_sibling();
123            }
124            None
125        });
126
127        container.add_controller(drag_source);
128
129        // DropTarget: accepts a source index and reorders within the parent Box.
130        let drop_target = gtk4::DropTarget::new(glib::Type::I32, gdk::DragAction::MOVE);
131
132        let drop_container = container.clone();
133        drop_target.connect_motion(|_target, _x, _y| gdk::DragAction::MOVE);
134
135        drop_target.connect_drop(move |_target, value, _x, _y| {
136            let Ok(source_index) = value.get::<i32>() else {
137                return false;
138            };
139
140            let Some(parent) = drop_container.parent().and_then(|p| p.downcast::<gtk4::Box>().ok()) else {
141                return false;
142            };
143
144            // Find target index by iterating siblings.
145            let mut target_index = 0i32;
146            let mut child = parent.first_child();
147            let mut found = false;
148            while let Some(c) = child {
149                if c == drop_container {
150                    found = true;
151                    break;
152                }
153                target_index += 1;
154                child = c.next_sibling();
155            }
156            if !found || source_index == target_index {
157                return false;
158            }
159
160            // Defer reorder to idle callback. Use reorder_child_after which
161            // moves the widget without remove/append (no reparenting, safe
162            // for GLArea/Dice3D widgets).
163            let parent = parent.clone();
164            glib::idle_add_local_once(move || {
165                // Collect all children to find source and target widgets.
166                let mut children: Vec<gtk4::Widget> = Vec::new();
167                let mut child = parent.first_child();
168                while let Some(c) = child {
169                    children.push(c.clone());
170                    child = c.next_sibling();
171                }
172
173                let source = source_index as usize;
174                let target = target_index as usize;
175                if source >= children.len() || target >= children.len() || source == target {
176                    return;
177                }
178
179                let source_widget = &children[source];
180                if source_index < target_index {
181                    // Moving down: insert after the target widget.
182                    let sibling = &children[target];
183                    parent.reorder_child_after(source_widget, Some(sibling));
184                } else {
185                    // Moving up: insert after the widget before target,
186                    // or prepend if target is 0.
187                    if target_index == 0 {
188                        parent.reorder_child_after(source_widget, None::<&gtk4::Widget>);
189                    } else {
190                        let sibling = &children[target - 1];
191                        parent.reorder_child_after(source_widget, Some(sibling));
192                    }
193                }
194            });
195
196            true
197        });
198
199        container.add_controller(drop_target);
200
201        Self { container }
202    }
203
204    /// Returns the root widget for packing.
205    pub fn widget(&self) -> &gtk4::Box {
206        &self.container
207    }
