1/// Vertex data for a 3D dice model.
2///
3/// Currently provides a hardcoded cube (D6) with positions, normals, UVs,
4/// face IDs, and indices. Each face has 4 vertices (2 triangles) with
5/// outward-facing normals. UVs map 0..1 per face for procedural pip
6/// rendering. Face IDs encode the die value (1-6) shown on each face.
7pub struct DiceModel {
8    /// Vertex positions: [x, y, z] per vertex.
9    pub positions: Vec<f32>,
10    /// Vertex normals: [x, y, z] per vertex.
11    pub normals: Vec<f32>,
12    /// UV coordinates: [u, v] per vertex (0..1 per face).
13    pub uvs: Vec<f32>,
14    /// Face IDs (die value 1-6) per vertex.
15    pub face_ids: Vec<f32>,
16    /// Triangle indices into the vertex arrays.
17    pub indices: Vec<u32>,
18}
19
20impl DiceModel {
21    /// Create a D6 cube model with per-face normals, UVs, and face IDs.
22    ///
23    /// The cube spans from -0.5 to +0.5 on each axis. Each face has
24    /// 4 vertices (2 triangles) with a distinct normal, so lighting
25    /// produces sharp edges between faces. Opposite faces sum to 7
26    /// (standard die convention): -X=1/+X=6, +Z=2/-Z=5, +Y=3/-Y=4.
27    pub fn d6() -> Self {
28        // 6 faces × 4 vertices = 24 vertices, 6 faces × 6 indices = 36 indices
29        // Each face: ([4 corners], normal, face_value)
30        let cube_faces: [([[f32; 3]; 4], [f32; 3], f32); 6] = [
31            // +X face → 6
32            ([[0.5, -0.5, -0.5], [0.5, 0.5, -0.5], [0.5, 0.5, 0.5], [0.5, -0.5, 0.5]], [1.0, 0.0, 0.0], 6.0),
33            // -X face → 1
34            ([[-0.5, -0.5, 0.5], [-0.5, 0.5, 0.5], [-0.5, 0.5, -0.5], [-0.5, -0.5, -0.5]], [-1.0, 0.0, 0.0], 1.0),
35            // +Y face (top) → 3
36            ([[-0.5, 0.5, 0.5], [0.5, 0.5, 0.5], [0.5, 0.5, -0.5], [-0.5, 0.5, -0.5]], [0.0, 1.0, 0.0], 3.0),
37            // -Y face (bottom) → 4
38            ([[-0.5, -0.5, -0.5], [0.5, -0.5, -0.5], [0.5, -0.5, 0.5], [-0.5, -0.5, 0.5]], [0.0, -1.0, 0.0], 4.0),
39            // +Z face → 2
40            ([[-0.5, -0.5, 0.5], [0.5, -0.5, 0.5], [0.5, 0.5, 0.5], [-0.5, 0.5, 0.5]], [0.0, 0.0, 1.0], 2.0),
41            // -Z face → 5
42            ([[0.5, -0.5, -0.5], [-0.5, -0.5, -0.5], [-0.5, 0.5, -0.5], [0.5, 0.5, -0.5]], [0.0, 0.0, -1.0], 5.0),
43        ];
44
45        // UV mapping per face corner (0..1 consistent for all faces)
46        let face_uvs: [[f32; 2]; 4] = [[0.0, 0.0], [1.0, 0.0], [1.0, 1.0], [0.0, 1.0]];
47
48        let mut positions = Vec::with_capacity(24 * 3);
49        let mut normals_out = Vec::with_capacity(24 * 3);
50        let mut uvs = Vec::with_capacity(24 * 2);
51        let mut face_ids = Vec::with_capacity(24);
52        let mut indices = Vec::with_capacity(36);
53
54        for (corners, normal, face_value) in &cube_faces {
55            let base = (positions.len() / 3) as u32;
56            for (i, corner) in corners.iter().enumerate() {
57                positions.extend_from_slice(corner);
58                normals_out.extend_from_slice(normal);
59                uvs.extend_from_slice(&face_uvs[i]);
60                face_ids.push(*face_value);
61            }
62            // Two triangles: (0,1,2) and (0,2,3)
63            indices.extend_from_slice(&[base, base + 1, base + 2, base, base + 2, base + 3]);
64        }
65
66        Self {
67            positions,
68            normals: normals_out,
69            uvs,
70            face_ids,
71            indices,
72        }
73    }
74}
