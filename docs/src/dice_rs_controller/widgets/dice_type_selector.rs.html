12/// Dropdown selector for choosing a dice shell type (D6, D20, etc.).
13///
14/// Displays isometric 3D dice icons in the dropdown items and persists
15/// the selected type to per-dice config on change.
16pub struct DiceTypeSelector {
17    dropdown: gtk4::DropDown,
18}
19
20impl DiceTypeSelector {
21    /// Create a new dice type selector for the given dice.
22    ///
23    /// The selector is initialized with the dice's current type and
24    /// wired to update the dice, 3D renderer, and config on change.
25    pub fn new(dice_service: &DiceService, dice_3d: &Dice3D) -> Self {
26        let dice_types = DiceType::sorted_by_count();
27        let dice_type_labels: Vec<String> = dice_types.iter().map(|t| t.to_string()).collect();
28        let dice_type_model = gtk4::StringList::new(&dice_type_labels.iter().map(|s| s.as_str()).collect::<Vec<_>>());
29        let dropdown = gtk4::DropDown::builder()
30            .model(&dice_type_model)
31            .tooltip_text("Dice shell type")
32            .css_classes(vec!["dice-type-selector"])
33            .build();
34
35        // Custom factory: render isometric dice icons in dropdown items.
36        let factory = gtk4::SignalListItemFactory::new();
37        let types_for_setup = dice_types.clone();
38        factory.connect_setup(move |_item, list_item| {
39            let overlay = gtk4::Overlay::builder().css_classes(vec!["dice-type-item"]).build();
40            list_item.set_child(Some(&overlay));
41        });
42        factory.connect_bind(move |_item, list_item| {
43            let position = list_item.position() as usize;
44            let dice_type = types_for_setup[position % types_for_setup.len()];
45            let icon = create_icon(dice_type);
46            icon.set_hexpand(true);
47            icon.set_vexpand(true);
48            icon.set_halign(gtk4::Align::Center);
49            icon.set_valign(gtk4::Align::Center);
50            let label = gtk4::Label::builder()
51                .label(dice_type.to_string())
52                .css_classes(vec!["dice-type-label"])
53                .halign(gtk4::Align::Center)
54                .valign(gtk4::Align::Center)
55                .build();
56            let overlay = list_item.child().and_downcast::<gtk4::Overlay>().expect("child is Overlay");
57            overlay.set_child(Some(&icon));
58            overlay.add_overlay(&label);
59        });
60        dropdown.set_factory(Some(&factory));
61
62        // Square size matching face display.
63        dropdown.set_size_request(80, 80);
64        dropdown.set_margin_top(8);
65        dropdown.set_margin_end(8);
66
67        // Select the current dice type.
68        let current_type = dice_service.dice().dice_type();
69        if let Some(pos) = dice_types.iter().position(|t| *t == current_type) {
70            dropdown.set_selected(pos as u32);
71        }
72
73        let dice_service_for_type = dice_service.clone();
74        let dice_3d_for_type = dice_3d.clone();
75        dropdown.connect_notify_local(Some("selected"), move |dd, _pspec| {
76            let Some(item) = dd.selected_item() else {
77                return;
78            };
79            let Some(text) = item.downcast::<gtk4::StringObject>().ok() else {
80                return;
81            };
82            match DiceType::from_str(text.string().as_str()) {
83                Ok(dt) => {
84                    dice_service_for_type.set_dice_type(dt);
85                    dice_3d_for_type.set_dice_type(dt);
86                }
87                Err(error) => debug!(error = %error, "invalid dice type selected"),
88            }
89        });
90
91        Self { dropdown }
92    }
93
94    /// Returns the dropdown widget for packing.
95    pub fn widget(&self) -> &gtk4::DropDown {
96        &self.dropdown
97    }
98}
99
100impl WidgetContainer for DiceTypeSelector {
101    fn widget(&self) -> &gtk4::Widget {
102        self.dropdown.as_ref()
103    }
