12/// Errors that can occur during OpenGL renderer initialization or rendering.
13#[derive(Debug, Error, Diagnostic)]
14pub enum RendererError {
15    #[error("failed to create OpenGL vertex array object")]
16    #[diagnostic(code(dice_renderer::vao_creation_failed))]
17    VaoCreationFailed,
18
19    #[error("failed to create OpenGL vertex buffer object")]
20    #[diagnostic(code(dice_renderer::vbo_creation_failed))]
21    VboCreationFailed,
22
23    #[error("failed to create OpenGL element buffer object")]
24    #[diagnostic(code(dice_renderer::ebo_creation_failed))]
25    EboCreationFailed,
26
27    #[error("failed to create OpenGL {shader_type} shader")]
28    #[diagnostic(code(dice_renderer::shader_creation_failed))]
29    ShaderCreationFailed { shader_type: String },
30
31    #[error("{shader_type} shader compilation failed: {log}")]
32    #[diagnostic(code(dice_renderer::shader_compilation_failed))]
33    ShaderCompilationFailed { shader_type: String, log: String },
34
35    #[error("failed to create OpenGL shader program")]
36    #[diagnostic(code(dice_renderer::program_creation_failed))]
37    ProgramCreationFailed,
38
39    #[error("shader program linking failed: {log}")]
40    #[diagnostic(code(dice_renderer::program_linking_failed))]
41    ProgramLinkingFailed { log: String },
42
43    #[error("shader attribute '{name}' not found in program")]
44    #[diagnostic(code(dice_renderer::attribute_not_found))]
45    AttributeNotFound { name: String },
46
47    #[error("shader uniform '{name}' not found in program")]
48    #[diagnostic(code(dice_renderer::uniform_not_found))]
49    UniformNotFound { name: String },
50}
51
52type Result<T> = std::result::Result<T, RendererError>;

54/// OpenGL renderer for 3D dice models using glow + glam.
55pub struct DiceRenderer {
56    pub gl: Rc<glow::Context>,
57    program: <glow::Context as HasContext>::Program,
58    vao: <glow::Context as HasContext>::VertexArray,
59    _vbo: <glow::Context as HasContext>::Buffer,
60    _ebo: <glow::Context as HasContext>::Buffer,
61    index_count: i32,
62    /// Light direction for diffuse lighting (normalized).
63    light_dir: Vec3,
64}

66const VERTEX_SHADER_SOURCE: &str = r#"#version 300 es
67precision highp float;
68in vec3 a_pos;
69in vec3 a_normal;
70in vec2 a_uv;
71in float a_face_id;
72uniform mat4 u_mvp;
73uniform mat4 u_model;
74out vec3 v_normal;
75out vec3 v_frag_pos;
76out vec2 v_uv;
77out float v_face_id;
78void main() {
79    v_normal = mat3(u_model) * a_normal;
80    v_frag_pos = vec3(u_model * vec4(a_pos, 1.0));
81    v_uv = a_uv;
82    v_face_id = a_face_id;
83    gl_Position = u_mvp * vec4(a_pos, 1.0);
84}
85"#;
86
87const FRAGMENT_SHADER_SOURCE: &str = r#"#version 300 es
88precision highp float;
89in vec3 v_normal;
90in vec3 v_frag_pos;
91in vec2 v_uv;
92in float v_face_id;
93uniform vec3 u_light_dir;
94uniform vec3 u_base_color;
95uniform vec3 u_edge_color;
96out vec4 frag_color;
97
98const float PIP_RADIUS = 0.07;
99const vec3 PIP_COLOR = vec3(0.02, 0.02, 0.02);
100
101bool in_pip(vec2 uv, vec2 center) {
102    return distance(uv, center) < PIP_RADIUS;
103}
104
105bool is_pip_fragment(float face_id, vec2 uv) {
106    int face = int(face_id + 0.5);
107    if (face == 1) {
108        return in_pip(uv, vec2(0.5, 0.5));
109    } else if (face == 2) {
110        return in_pip(uv, vec2(0.25, 0.25)) || in_pip(uv, vec2(0.75, 0.75));
111    } else if (face == 3) {
112        return in_pip(uv, vec2(0.25, 0.25)) || in_pip(uv, vec2(0.5, 0.5)) || in_pip(uv, vec2(0.75, 0.75));
113    } else if (face == 4) {
114        return in_pip(uv, vec2(0.25, 0.25)) || in_pip(uv, vec2(0.75, 0.25))
115            || in_pip(uv, vec2(0.25, 0.75)) || in_pip(uv, vec2(0.75, 0.75));
116    } else if (face == 5) {
117        return in_pip(uv, vec2(0.25, 0.25)) || in_pip(uv, vec2(0.75, 0.25))
118            || in_pip(uv, vec2(0.5, 0.5))
119            || in_pip(uv, vec2(0.25, 0.75)) || in_pip(uv, vec2(0.75, 0.75));
120    } else if (face == 6) {
121        return in_pip(uv, vec2(0.25, 0.2)) || in_pip(uv, vec2(0.75, 0.2))
122            || in_pip(uv, vec2(0.25, 0.5)) || in_pip(uv, vec2(0.75, 0.5))
123            || in_pip(uv, vec2(0.25, 0.8)) || in_pip(uv, vec2(0.75, 0.8));
124    }
125    return false;
126}
127
128void main() {
129    vec3 normal = normalize(v_normal);
130    float diff = max(dot(normal, normalize(u_light_dir)), 0.0);
131    vec3 base = is_pip_fragment(v_face_id, v_uv) ? PIP_COLOR : u_base_color;
132
133    // Render face edges with a distinct color to make the cube shape visible.
134    float edge_dist = min(min(v_uv.x, v_uv.y), min(1.0 - v_uv.x, 1.0 - v_uv.y));
135    if (edge_dist < 0.08 && !is_pip_fragment(v_face_id, v_uv)) {
136        base = u_edge_color;
137    }
138
139    vec3 ambient = base * 0.35;
140    vec3 diffuse = base * diff * 0.65;
141    frag_color = vec4(ambient + diffuse, 1.0);
142}
143"#;

145impl DiceRenderer {
146    /// Create a new renderer with the given GL context.
147    pub fn new(gl: Rc<glow::Context>, model: &DiceModel) -> Result<Self> {
148        let program = unsafe { Self::compile_program(&gl)? };
149
150        let vao = unsafe { gl.create_vertex_array().map_err(|_| RendererError::VaoCreationFailed)? };
151        let vbo = unsafe { gl.create_buffer().map_err(|_| RendererError::VboCreationFailed)? };
152        let ebo = unsafe { gl.create_buffer().map_err(|_| RendererError::EboCreationFailed)? };
153
154        unsafe {
155            gl.bind_vertex_array(Some(vao));
156
157            // Interleaved vertex data: pos(3) + normal(3) + uv(2) + face_id(1) = 9 floats per vertex
158            let stride = 9 * std::mem::size_of::<f32>() as i32;
159            let vertex_count = model.positions.len() / 3;
160            let mut vertex_data: Vec<f32> = Vec::with_capacity(vertex_count * 9);
161            for i in 0..vertex_count {
162                vertex_data.extend_from_slice(&model.positions[i * 3..i * 3 + 3]);
163                vertex_data.extend_from_slice(&model.normals[i * 3..i * 3 + 3]);
164                vertex_data.extend_from_slice(&model.uvs[i * 2..i * 2 + 2]);
165                vertex_data.push(model.face_ids[i]);
166            }
167
168            gl.bind_buffer(glow::ARRAY_BUFFER, Some(vbo));
169            gl.buffer_data_u8_slice(glow::ARRAY_BUFFER, vertex_data.align_to::<u8>().1, glow::STATIC_DRAW);
170
171            gl.bind_buffer(glow::ELEMENT_ARRAY_BUFFER, Some(ebo));
172            let index_data: &[u8] = bytemuck::cast_slice(&model.indices);
173            gl.buffer_data_u8_slice(glow::ELEMENT_ARRAY_BUFFER, index_data, glow::STATIC_DRAW);
174
175            let pos_loc = gl
176                .get_attrib_location(program, "a_pos")
177                .ok_or(RendererError::AttributeNotFound { name: "a_pos".to_string() })?;
178            gl.enable_vertex_attrib_array(pos_loc);
179            gl.vertex_attrib_pointer_f32(pos_loc, 3, glow::FLOAT, false, stride, 0);
180
181            let normal_loc = gl
182                .get_attrib_location(program, "a_normal")
183                .ok_or(RendererError::AttributeNotFound { name: "a_normal".to_string() })?;
184            gl.enable_vertex_attrib_array(normal_loc);
185            gl.vertex_attrib_pointer_f32(normal_loc, 3, glow::FLOAT, false, stride, 3 * std::mem::size_of::<f32>() as i32);
186
187            let uv_loc = gl
188                .get_attrib_location(program, "a_uv")
189                .ok_or(RendererError::AttributeNotFound { name: "a_uv".to_string() })?;
190            gl.enable_vertex_attrib_array(uv_loc);
191            gl.vertex_attrib_pointer_f32(uv_loc, 2, glow::FLOAT, false, stride, 6 * std::mem::size_of::<f32>() as i32);
192
193            let face_id_loc = gl
194                .get_attrib_location(program, "a_face_id")
195                .ok_or(RendererError::AttributeNotFound { name: "a_face_id".to_string() })?;
196            gl.enable_vertex_attrib_array(face_id_loc);
197            gl.vertex_attrib_pointer_f32(face_id_loc, 1, glow::FLOAT, false, stride, 8 * std::mem::size_of::<f32>() as i32);
198
199            gl.bind_vertex_array(None);
200        }
201
202        Ok(Self {
203            gl,
204            program,
205            vao,
206            _vbo: vbo,
207            _ebo: ebo,
208            index_count: model.indices.len() as i32,
209            light_dir: Vec3::new(0.5, -1.0, 0.3).normalize(),
210        })
211    }
212
213    /// Render the dice model with the given orientation and aspect ratio.
214    #[allow(deprecated)]
215    pub fn render(&self, orientation: Quat, aspect: f32, base_color: [f32; 3], edge_color: [f32; 3]) -> Result<()> {
216        let gl = &self.gl;
217
218        let model_matrix = Mat4::from_quat(orientation);
219        let eye = Vec3::new(2.0, 2.0, 3.0);
220        let target = Vec3::ZERO;
221        let view_matrix = Mat4::look_at_rh(eye, target, Vec3::Y);
222        let projection = Mat4::perspective_rh(std::f32::consts::FRAC_PI_4, aspect, 0.1, 100.0);
223        let mvp = projection * view_matrix * model_matrix;
224
225        unsafe {
226            gl.use_program(Some(self.program));
227
228            let mvp_loc = gl
229                .get_uniform_location(self.program, "u_mvp")
230                .ok_or(RendererError::UniformNotFound { name: "u_mvp".to_string() })?;
231            gl.uniform_matrix_4_f32_slice(Some(&mvp_loc), false, &mvp.to_cols_array());
232
233            let model_loc = gl
234                .get_uniform_location(self.program, "u_model")
235                .ok_or(RendererError::UniformNotFound { name: "u_model".to_string() })?;
236            gl.uniform_matrix_4_f32_slice(Some(&model_loc), false, &model_matrix.to_cols_array());
237
238            let light_loc = gl.get_uniform_location(self.program, "u_light_dir").ok_or(RendererError::UniformNotFound {
239                name: "u_light_dir".to_string(),
240            })?;
241            gl.uniform_3_f32(Some(&light_loc), self.light_dir.x, self.light_dir.y, self.light_dir.z);
242
243            let color_loc = gl.get_uniform_location(self.program, "u_base_color").ok_or(RendererError::UniformNotFound {
244                name: "u_base_color".to_string(),
245            })?;
246            gl.uniform_3_f32(Some(&color_loc), base_color[0], base_color[1], base_color[2]);
247
248            let edge_loc = gl.get_uniform_location(self.program, "u_edge_color").ok_or(RendererError::UniformNotFound {
249                name: "u_edge_color".to_string(),
250            })?;
251            gl.uniform_3_f32(Some(&edge_loc), edge_color[0], edge_color[1], edge_color[2]);
252
253            gl.enable(glow::DEPTH_TEST);
254            gl.enable(glow::CULL_FACE);
255            gl.cull_face(glow::BACK);
256
257            gl.bind_vertex_array(Some(self.vao));
258            gl.draw_elements(glow::TRIANGLES, self.index_count, glow::UNSIGNED_INT, 0);
259            gl.bind_vertex_array(None);
260        }
261
262        Ok(())
263    }
264
265    unsafe fn compile_program(gl: &Rc<glow::Context>) -> Result<<glow::Context as HasContext>::Program> {
266        unsafe {
267            let vertex_shader = gl.create_shader(glow::VERTEX_SHADER).map_err(|_| RendererError::ShaderCreationFailed {
268                shader_type: "vertex".to_string(),
269            })?;
270            gl.shader_source(vertex_shader, VERTEX_SHADER_SOURCE);
271            gl.compile_shader(vertex_shader);
272            if !gl.get_shader_compile_status(vertex_shader) {
273                let log = gl.get_shader_info_log(vertex_shader);
274                return Err(RendererError::ShaderCompilationFailed {
275                    shader_type: "vertex".to_string(),
276                    log,
277                });
278            }
279
280            let fragment_shader = gl.create_shader(glow::FRAGMENT_SHADER).map_err(|_| RendererError::ShaderCreationFailed {
281                shader_type: "fragment".to_string(),
282            })?;
283            gl.shader_source(fragment_shader, FRAGMENT_SHADER_SOURCE);
284            gl.compile_shader(fragment_shader);
285            if !gl.get_shader_compile_status(fragment_shader) {
286                let log = gl.get_shader_info_log(fragment_shader);
287                return Err(RendererError::ShaderCompilationFailed {
288                    shader_type: "fragment".to_string(),
289                    log,
290                });
291            }
292
293            let program = gl.create_program().map_err(|_| RendererError::ProgramCreationFailed)?;
294            gl.attach_shader(program, vertex_shader);
295            gl.attach_shader(program, fragment_shader);
296            gl.link_program(program);
297            if !gl.get_program_link_status(program) {
298                let log = gl.get_program_info_log(program);
299                return Err(RendererError::ProgramLinkingFailed { log });
300            }
301
302            gl.delete_shader(vertex_shader);
303            gl.delete_shader(fragment_shader);
304
305            Ok(program)
306        }
307    }
