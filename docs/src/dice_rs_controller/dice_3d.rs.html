18/// Orientation state for smooth interpolation.
19#[derive(Clone)]
20struct OrientationState {
21    orientation: Quat,
22    target: Quat,
23    color: [f32; 3],
24    spin_angle: f32,
25}
26
27impl Default for OrientationState {
28    fn default() -> Self {
29        Self {
30            orientation: Quat::IDENTITY,
31            target: Quat::IDENTITY,
32            color: [0.95, 0.95, 0.95],
33            spin_angle: 0.0,
34        }
35    }
36}

38/// GTK4 widget that renders a 3D dice model using OpenGL (glow + glam).
39///
40/// Wraps a `gtk4::GLArea` without subclassing - connects to signals directly.
41#[derive(Clone)]
42pub struct Dice3D {
43    gl_area: gtk4::GLArea,
44    orientation: Rc<RefCell<OrientationState>>,
45    renderer: Rc<RefCell<Option<DiceRenderer>>>,
46}
47
48impl Dice3D {
49    /// Create a new 3D dice widget.
50    pub fn new() -> Self {
51        let gl_area = gtk4::GLArea::builder().has_depth_buffer(true).auto_render(true).build();
52
53        let widget = Self {
54            gl_area,
55            orientation: Rc::new(RefCell::new(OrientationState::default())),
56            renderer: Rc::new(RefCell::new(None)),
57        };
58
59        widget.connect_signals();
60        widget
61    }
62
63    /// Set the target orientation from accelerometer data.
64    pub fn set_orientation(&self, acceleration: Acceleration) {
65        let gravity = Vec3::new(acceleration.x as f32, acceleration.y as f32, acceleration.z as f32);
66
67        if gravity.length() < 0.1 {
68            return;
69        }
70
71        let gravity = gravity.normalize();
72        let target = Quat::from_rotation_arc(gravity, Vec3::Y);
73
74        self.orientation.borrow_mut().target = target;
75        self.gl_area.queue_render();
76    }
77
78    /// Set the target orientation directly from a quaternion.
79    #[allow(dead_code)]
80    pub fn set_orientation_quat(&self, quat: Quat) {
81        self.orientation.borrow_mut().target = quat;
82        self.gl_area.queue_render();
83    }
84
85    /// Set the dice color for rendering.
86    pub fn set_color(&self, color: DiceColor) {
87        let rgb = match color {
88            DiceColor::Black => [0.2, 0.2, 0.2],
89            DiceColor::Red => [0.8, 0.15, 0.15],
90            DiceColor::Green => [0.15, 0.7, 0.2],
91            DiceColor::Blue => [0.15, 0.3, 0.85],
92            DiceColor::Yellow => [0.9, 0.8, 0.1],
93            DiceColor::Orange => [0.9, 0.5, 0.1],
94        };
95        self.orientation.borrow_mut().color = rgb;
96        self.gl_area.queue_render();
97    }
98
99    /// Returns the underlying GLArea widget for packing.
100    pub fn widget(&self) -> &gtk4::GLArea {
101        &self.gl_area
102    }
103
104    fn connect_signals(&self) {
105        let orientation = self.orientation.clone();
106        let renderer = self.renderer.clone();
107        let gl_area = self.gl_area.clone();
108
109        self.gl_area.connect_render(move |area, _context| {
110            // Initialize renderer on first render.
111            if renderer.borrow().is_none() {
112                area.make_current();
113
114                if area.error().is_some() {
115                    return gtk4::glib::Propagation::Proceed;
116                }
117
118                gl_loader::init_gl();
119                let loader = |sym: &str| -> *const std::ffi::c_void { gl_loader::get_proc_address(sym) as *const std::ffi::c_void };
120
121                let glow_ctx = unsafe { glow::Context::from_loader_function(loader) };
122                let model = DiceModel::d6();
123                match DiceRenderer::new(Rc::new(glow_ctx), &model) {
124                    Ok(r) => *renderer.borrow_mut() = Some(r),
125                    Err(e) => {
126                        error!(error = %miette::Report::new(e), "failed to initialize dice renderer");
127                        return gtk4::glib::Propagation::Proceed;
128                    }
129                }
130            }
131
132            let r = renderer.borrow();
133            let Some(renderer) = r.as_ref() else {
134                return gtk4::glib::Propagation::Proceed;
135            };
136
137            let width = area.width() as f32;
138            let height = area.height() as f32;
139            let aspect = if height > 0.0 { width / height } else { 1.0 };
140
141            // Snap to target orientation, then apply continuous Y-axis spin.
142            let mut state = orientation.borrow_mut();
143            state.spin_angle += 0.02;
144            let spin = Quat::from_rotation_y(state.spin_angle);
145            let render_orientation = spin * state.target;
146            state.orientation = render_orientation;
147            drop(state);
148
149            unsafe {
150                renderer.gl.clear_color(0.1, 0.1, 0.12, 1.0);
151                renderer.gl.clear(glow::COLOR_BUFFER_BIT | glow::DEPTH_BUFFER_BIT);
152            }
153
154            renderer
155                .render(render_orientation, aspect, [0.95, 0.95, 0.95], orientation.borrow().color)
156                .unwrap_or_else(|e| {
157                    error!(error = %miette::Report::new(e), "dice render failed");
158                });
159
160            gtk4::glib::Propagation::Proceed
161        });
162
163        // Drive continuous rendering with a timer for smooth animation.
164        let timer_area = gl_area.clone();
165        glib::timeout_add_local(Duration::from_millis(16), move || {
166            timer_area.queue_render();
167            glib::ControlFlow::Continue
168        });
169    }
170}
