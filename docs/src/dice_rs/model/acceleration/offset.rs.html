1/// Software calibration offset computed from a resting accelerometer sample.
2///
3/// When the firmware does not support hardware calibration via BLE,
4/// `calibrate_software()` captures the current XYZ reading and computes
5/// the deviation from the expected gravity vector. The offset is subtracted
6/// from all subsequent accelerometer readings before face value interpretation.
7#[derive(Debug, Clone, Copy, PartialEq, Default)]
8pub struct AccelerationOffset {
9    /// X-axis deviation from the expected resting vector.
10    pub dx: i8,
11    /// Y-axis deviation from the expected resting vector.
12    pub dy: i8,
13    /// Z-axis deviation from the expected resting vector.
14    pub dz: i8,
15}
