16/// Default name prefix for GoDice devices.
17const DEFAULT_NAME_PREFIX: &str = "GoDice_";
18
19/// Default scan duration.
20const DEFAULT_SCAN_DURATION: Duration = Duration::from_secs(5);
21
22/// Scans for GoDice BLE devices in range.
23pub struct DiceScanner<T: BleTransport> {
24    transport: Arc<T>,
25    name_prefix: String,
26    scan_duration: Duration,
27}
28
29impl<T: BleTransport> DiceScanner<T> {
30    /// Create a scanner with the default prefix "GoDice_" and 5s scan duration.
31    pub fn new(transport: Arc<T>) -> Self {
32        Self {
33            transport,
34            name_prefix: DEFAULT_NAME_PREFIX.to_string(),
35            scan_duration: DEFAULT_SCAN_DURATION,
36        }
37    }
38
39    /// Set a custom name prefix.

41        Self {
42            name_prefix: prefix.into(),
43            ..self
44        }
45    }
46
47    /// Set a custom scan duration.
48    pub fn with_scan_duration(self, duration: Duration) -> Self {
49        Self {
50            scan_duration: duration,
51            ..self
52        }
53    }
54
55    /// Scan for GoDice devices. Returns when the scan duration elapses or
56    /// the transport stops scanning.
57    ///
58    /// Consumes the `CentralEvent` stream during the scan to collect RSSI
59    /// values from `RssiUpdate` events. Multiple RSSI samples per device are
60    /// averaged for a more stable value. Falls back to `read_rssi()` or
61    /// cached `properties().rssi` if no events were received.
62    ///
63    /// Uses a two-stage filter:
64    /// 1. `ScanFilter` with the NUS service UUID (if supported by platform).
65    /// 2. Fallback: filter `PeripheralProperties::local_name` by prefix.
66    pub async fn scan(&self) -> Result<Vec<DiceDevice>> {
67        self.transport.start_scan(ScanFilter::default()).await?;
68
69        let mut events = self.transport.events().await?;
70        let mut rssi_samples: HashMap<btleplug::platform::PeripheralId, Vec<i16>> = HashMap::new();
71
72        let scan_deadline = tokio::time::sleep(self.scan_duration);
73        tokio::pin!(scan_deadline);
74
75        loop {
76            select! {
77                _ = &mut scan_deadline => break,
78                event = events.next() => {
79                    if let Some(CentralEvent::RssiUpdate { id, rssi }) = event {
80                        rssi_samples.entry(id).or_default().push(rssi);
81                    }
82                }
83            }
84        }
85
86        self.transport.stop_scan().await?;
87
88        let peripherals = self.transport.peripherals().await?;
89        let mut devices = Vec::new();
90
91        for peripheral in peripherals {
92            let props = peripheral.properties().await?;
93            if let Some(props) = props {
94                let local_name = props.local_name.as_deref().unwrap_or("");
95                if local_name.starts_with(&self.name_prefix) {
96                    let id = peripheral.id();
97                    let avg_rssi = rssi_samples.get(&id).filter(|samples| !samples.is_empty()).map(|samples| {
98                        let sum: i64 = samples.iter().map(|r| *r as i64).sum();
99                        (sum / samples.len() as i64) as i16
100                    });
101                    let avg_rssi = match avg_rssi {
102                        Some(rssi) => Some(rssi),
103                        None => match peripheral.read_rssi().await {
104                            Ok(rssi) => Some(rssi),
105                            Err(_) => props.rssi,
106                        },
107                    };
108
109                    debug!(name = local_name, rssi = ?avg_rssi, "found GoDice device");
110                    devices.push(DiceDevice {
111                        id,
112                        address: peripheral.address(),
113                        name: local_name.to_string(),
114                        rssi: avg_rssi,
115                    });
116                }
117            }
118        }
119
120        Ok(devices)
121    }
122}
