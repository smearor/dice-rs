20/// Trait abstracting dice manager operations for testability.
21///
22/// Allows `dice-rs-ws` to hold a trait object (`Arc<dyn DiceService>`)
23/// instead of a concrete `DiceManager`, enabling mock implementations
24/// in integration tests without a real BLE adapter.
25#[async_trait]
26pub trait DiceService: Send + Sync {
27    /// Scan for GoDice devices using the default scan duration.
28    async fn scan(&self) -> Result<Vec<DiceDevice>>;
29
30    /// Scan for GoDice devices with a custom scan duration.
31    async fn scan_with_duration(&self, duration: Duration) -> Result<Vec<DiceDevice>>;
32
33    /// Connect to a discovered device.
34    async fn connect(&self, device: &DiceDevice) -> Result<Dice>;
35
36    /// Find a discovered device by MAC address (partial match).
37    ///
38    /// Scans for devices and returns the first whose address contains the
39    /// given substring. Useful for matching a short MAC prefix.
40    async fn find_device_by_address(&self, address: &str) -> Result<DiceDevice>;
41}

48impl DiceManager {
49    /// Create a new manager. Internally calls `Manager::new()` and selects
50    /// the first available Bluetooth adapter.
51    pub async fn new() -> Result<Self> {
52        let transport = BtleplugTransport::new().await?;
53        Ok(Self {
54            transport: Arc::new(transport),
55        })
56    }

58    /// Create a scanner for discovering GoDice devices.
59    pub fn scanner(&self) -> DiceScanner<BtleplugTransport> {
60        DiceScanner::new(self.transport.clone())
61    }
62
63    /// Scan for GoDice devices using the default scanner settings.
64    pub async fn scan(&self) -> Result<Vec<DiceDevice>> {
65        self.scanner().scan().await
66    }
67
68    /// Scan for GoDice devices with a custom scan duration.
69    pub async fn scan_with_duration(&self, duration: Duration) -> Result<Vec<DiceDevice>> {
70        self.scanner().with_scan_duration(duration).scan().await
71    }

73    /// Find a discovered device by MAC address (partial match).
74    ///
75    /// Scans for devices and returns the first whose address contains the
76    /// given substring. Useful for matching a short MAC prefix.
77    pub async fn find_device_by_address(&self, address: &str) -> Result<DiceDevice> {
78        let devices = self.scan().await?;
79        devices
80            .into_iter()
81            .find(|d| d.address.to_string().contains(address))
82            .ok_or_else(|| BleError::device_not_found(address).into())
83    }
84
85    /// Connect to a dice by MAC address (scans first).
86    ///
87    /// Convenience method combining `find_device_by_address` and `connect`.
88    pub async fn connect_by_address(&self, address: &str) -> Result<Dice> {
89        let device = self.find_device_by_address(address).await?;
90        self.connect(&device).await
91    }

93    /// Connect to a discovered device.
94    ///
95    /// Performs:
96    /// 1. `peripheral.connect()` (with up to 3 retries and 1s backoff)
97    /// 2. `peripheral.discover_services()`
98    /// 3. Find write char (`6e400002`) and notify char (`6e400003`)
99    /// 4. `peripheral.subscribe(notify_char)`
100    /// 5. `peripheral.notifications()` → spawn parse task
101    /// 6. Return `Dice` handle
102    ///
103    /// The retry loop handles transient connection failures (e.g. a GoDice
104    /// that is advertising but not yet accepting connections while charging
105    /// from 0% battery).
106    pub async fn connect(&self, device: &DiceDevice) -> Result<Dice> {
107        let peripherals = self.transport.peripherals().await?;
108        let peripheral = peripherals
109            .into_iter()
110            .find(|p| p.id() == device.id)
111            .ok_or_else(|| DiceError::from(BleError::peripheral_not_found(&device.name)))?;
112
113        let max_retries = 3;
114        let backoff = Duration::from_secs(1);
115        let mut last_error: DiceError = BleError::NoAttemptMade.into();
116        for attempt in 0..max_retries {
117            match peripheral.connect().await {
118                Ok(()) => break,
119                Err(error) => {
120                    last_error = error;
121                    debug!(attempt, %last_error, device = %device.name, "connect attempt failed, retrying");
122                    if attempt + 1 < max_retries {
123                        tokio::time::sleep(backoff).await;
124                    }
125                }
126            }
127        }
128        if !peripheral.is_connected().await? {
129            return Err(last_error);
130        }
131
132        peripheral.discover_services().await?;
133
134        let write_char = peripheral
135            .characteristic(NUS_WRITE_CHAR_UUID)
136            .ok_or_else(|| DiceError::from(BleError::characteristic_not_found(NusCharacteristic::Write)))?;
137        let notify_char = peripheral
138            .characteristic(NUS_NOTIFY_CHAR_UUID)
139            .ok_or_else(|| DiceError::from(BleError::characteristic_not_found(NusCharacteristic::Notify)))?;
140
141        peripheral.subscribe(&notify_char).await?;
142
143        let dice = Dice::new(peripheral, device.name.clone(), write_char, notify_char);
144        dice.spawn_notification_task().await?;
145        dice.spawn_led_debounce_task();
146        dice.spawn_connection_monitor();
147
148        dice.init().await?;
149
150        Ok(dice)
151    }

153    /// Attempt to reconnect to a disconnected dice.
154    ///
155    /// Retries with exponential backoff until success or max retries.
156    /// Mirrors the JS API's `attemptReconnect` behavior.
157    pub async fn reconnect(&self, dice: &Dice) -> Result<()> {
158        let mut backoff = Duration::from_millis(500);
159        let max_backoff = Duration::from_secs(5);
160        let max_retries = 10;
161
162        for attempt in 0..max_retries {
163            if dice.is_connected().await? {
164                return Ok(());
165            }
166            if let Err(error) = dice.reconnect_internal().await {
167                debug!(attempt, %error, "reconnect attempt failed");
168            }
169            tokio::time::sleep(backoff).await;
170            backoff = (backoff * 2).min(max_backoff);
171        }
172        Err(BleError::ReconnectFailed.into())
173    }
174
175    /// Disconnect all dice and release the BLE adapter.
176    pub async fn shutdown(&self) -> Result<()> {
177        self.transport.stop_scan().await
178    }
179
180    /// Disconnect a peripheral by MAC address.
181    ///
182    /// Scans for peripherals and disconnects the one matching the given address.
183    /// Useful for cleaning up stale connections before scanning, since BlueZ
184    /// does not deliver RSSI advertisement updates for connected devices.
185    pub async fn disconnect_by_address(&self, address: &str) -> Result<()> {
186        let peripherals = self.transport.peripherals().await?;
187        for peripheral in peripherals {
188            if peripheral.address().to_string().contains(address) && peripheral.is_connected().await? {
189                debug!(address = %peripheral.address(), "disconnecting peripheral");
190                peripheral.disconnect().await?;
191            }
192        }
193        Ok(())
194    }
195
196    /// Disconnect all connected GoDice devices.
197    ///
198    /// Iterates all known peripherals, filters by the GoDice name prefix,
199    /// and disconnects those that are currently connected. Returns the
200    /// number of devices that were disconnected.
201    pub async fn disconnect_all(&self) -> Result<usize> {
202        let peripherals = self.transport.peripherals().await?;
203        let mut count = 0;
204        for peripheral in peripherals {
205            let props = peripheral.properties().await?;
206            if let Some(props) = props {
207                let local_name = props.local_name.as_deref().unwrap_or("");
208                if local_name.starts_with("GoDice_") && peripheral.is_connected().await? {
209                    debug!(name = local_name, address = %peripheral.address(), "disconnecting GoDice");
210                    peripheral.disconnect().await?;
211                    count += 1;
212                }
213            }
214        }
215        Ok(count)
216    }
