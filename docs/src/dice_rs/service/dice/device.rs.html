16/// A discovered GoDice device, not yet connected.
17#[derive(Debug, Clone, Serialize, Deserialize)]
18pub struct DiceDevice {
19    /// Unique BLE identifier (not serialized).
20    #[serde(skip, default = "default_peripheral_id")]
21    pub id: PeripheralId,
22    /// MAC address.
23    #[serde(serialize_with = "serialize_bdaddr", deserialize_with = "deserialize_bdaddr")]
24    pub address: BDAddr,
25    /// Advertised device name (e.g. "GoDice_001234").
26    pub name: String,
27    /// Received signal strength indicator (if available).
28    pub rssi: Option<i16>,
29}
30
31impl DiceDevice {
32    /// Extract the physical dice color from the device name.
33    ///
34    /// Name format: `GoDice_{HEXID}_{COLOR}_v{VERSION}` where COLOR is a
35    /// single letter: K=Black, R=Red, G=Green, B=Blue, Y=Yellow, O=Orange.
36    ///
37    /// Returns `DiceError::InvalidColor` if the color code cannot be parsed.
38    pub fn color(&self) -> Result<DiceColor, DiceError> {
39        let parts: Vec<&str> = self.name.split('_').collect();
40        let code = parts.get(parts.len().saturating_sub(2)).ok_or(DiceError::InvalidColor(0))?;
41        let ch = code.chars().next().ok_or(DiceError::InvalidColor(0))?;
42        DiceColor::try_from(ch).map_err(|_| DiceError::InvalidColor(ch as u8))
43    }
44}
45
46impl Display for DiceDevice {
47    fn fmt(&self, f: &mut Formatter<'_>) -> std::fmt::Result {
48        let color = self.color().map(|c| c.to_string()).unwrap_or_else(|_| "Unknown".into());
49        let rssi = self.rssi.map(|r| format!("{r}")).unwrap_or_else(|| "N/A".into());
50        write!(f, "{} {} {} {}", self.address, self.name, color, rssi)
51    }
52}

54/// Produce a sentinel `PeripheralId` for deserialization contexts where the real
55/// BLE identifier is not available (e.g. JSON from a client).
56///
57/// `PeripheralId` wraps a `DeviceId` with no public constructor, so we use
58/// `MaybeUninit::zeroed` to create an inert placeholder. The `id` field is
59/// only meaningful for live BLE operations and is never sent to clients.
60fn default_peripheral_id() -> PeripheralId {
61    // SAFETY: `PeripheralId` is a thin wrapper around `DeviceId` which contains
62    // a `dbus::Path<'static>` (a `String`-like type). Zeroing produces an
63    // empty path, which is a valid but meaningless value. This placeholder is
64    // only used when deserializing `DiceDevice` from JSON where `id` was
65    // skipped; it should never be used to address a real BLE device.
66    unsafe { MaybeUninit::zeroed().assume_init() }
67}

69/// Serialize a `BDAddr` as a colon-separated MAC string (e.g. "AA:BB:CC:DD:EE:FF").
70fn serialize_bdaddr<S>(addr: &BDAddr, serializer: S) -> Result<S::Ok, S::Error>
71where
72    S: Serializer,
73{
74    let bytes = addr.as_ref();
75    let hex: String = bytes.iter().map(|b| format!("{b:02X}")).collect::<Vec<_>>().join(":");
76    serializer.serialize_str(&hex)
77}

79/// Deserialize a `BDAddr` from a MAC string (e.g. "AA:BB:CC:DD:EE:FF").
80fn deserialize_bdaddr<'de, D>(deserializer: D) -> Result<BDAddr, D::Error>
81where
82    D: Deserializer<'de>,
83{
84    let s = String::deserialize(deserializer)?;
85    let bytes: Vec<u8> = s
86        .split(':')
87        .map(|part| u8::from_str_radix(part, 16).map_err(serde::de::Error::custom))
88        .collect::<Result<Vec<_>, _>>()?;
89    if bytes.len() != 6 {
90        return Err(serde::de::Error::custom("BDAddr must be 6 bytes"));
91    }
92    let arr: [u8; 6] = bytes.as_slice().try_into().map_err(serde::de::Error::custom)?;
93    Ok(BDAddr::from(arr))
94}
