1use crate::model::acceleration::Acceleration;
2use crate::model::charging_state::ChargingState;

8/// High-level events emitted by a connected GoDice.
9#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
10#[serde(tag = "kind")]
11pub enum DiceEvent {
12    /// Dice has started rolling.
13    RollStart,
14    /// Dice is stable and flat after a roll.
15    Stable { face: FaceValue, acceleration: Acceleration },
16    /// Dice is stable but tilted after a roll.
17    TiltStable { face: FaceValue, acceleration: Acceleration },
18    /// Dice is stable after a fake roll.
19    FakeStable { face: FaceValue, acceleration: Acceleration },
20    /// Dice is stable after a small movement (face rotation).
21    MoveStable { face: FaceValue, acceleration: Acceleration },
22    /// Dice charging status changed.
23    Charging { state: ChargingState },
24    /// Single tap detected.
25    Tap,
26    /// Double tap detected.
27    DoubleTap,
28    /// Dice has disconnected.
29    Disconnected,
30}
31
32impl DiceEvent {
33    /// Returns the stability descriptor for stable/rolling events.
34    pub fn stability(&self) -> Option<StabilityDescriptor> {
35        match self {
36            Self::RollStart => Some(StabilityDescriptor::Rolling),
37            Self::Stable { .. } => Some(StabilityDescriptor::Stable),
38            Self::TiltStable { .. } => Some(StabilityDescriptor::TiltStable),
39            Self::FakeStable { .. } => Some(StabilityDescriptor::FakeStable),
40            Self::MoveStable { .. } => Some(StabilityDescriptor::MoveStable),
41            Self::Charging { .. } => None,
42            Self::Tap => None,
43            Self::DoubleTap => None,
44            Self::Disconnected => None,
45        }
46    }
47}
48
49impl std::fmt::Display for DiceEvent {
50    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
51        match self {
52            Self::RollStart => write!(f, "rolling"),
53            Self::Stable { face, .. } => write!(f, "stable face={face}"),
54            Self::TiltStable { face, .. } => write!(f, "tilt-stable face={face}"),
55            Self::FakeStable { face, .. } => write!(f, "fake-stable face={face}"),
56            Self::MoveStable { face, .. } => write!(f, "move-stable face={face}"),
57            Self::Charging { state } => write!(f, "charging={state}"),
58            Self::Tap => write!(f, "tap"),
59            Self::DoubleTap => write!(f, "double-tap"),
60            Self::Disconnected => write!(f, "disconnected"),
61        }
62    }
63}
