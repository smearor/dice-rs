19/// Internal shared state for a connected dice.
20/// Stored behind `Arc` so all `Dice` clones share the same state.
21pub struct DiceInner {
22    /// Advertised device name (e.g. "GoDice_7D8E7D_O_v04").
23    pub name: String,
24    /// BLE peripheral for write/subscribe operations.
25    pub peripheral: BtleplugPeripheralWrapper,
26    /// Write characteristic (NUS RX).
27    pub write_char: Characteristic,
28    /// Notify characteristic (NUS TX).
29    pub notify_char: Characteristic,
30    /// Broadcast sender for `DiceEvent` stream.
31    pub event_sender: broadcast::Sender<DiceEvent>,
32    /// Current dice type stored as `AtomicU8` for lock-free reads.
33    /// Converted to `DiceType` via `TryFrom<u8>` at use site.
34    /// This avoids async lock overhead in the notification task hot path.
35    /// `Arc`-wrapped so it can be cloned into the notification task.
36    pub dice_type: Arc<AtomicU8>,
37    /// FIFO queue of pending battery level request senders.
38    pub pending_battery: Arc<Mutex<VecDeque<Sender<u8>>>>,
39    /// FIFO queue of pending dice color request senders.
40    pub pending_color: Arc<Mutex<VecDeque<Sender<DiceColor>>>>,
41    /// FIFO queue of pending calibration request senders.
42    pub pending_calibration: Arc<Mutex<VecDeque<Sender<bool>>>>,
43    /// JoinHandle of the notification parsing task.
44    /// Aborted on disconnect/reconnect to prevent orphaned tasks.
45    pub notification_handle: Mutex<Option<JoinHandle<()>>>,
46    /// JoinHandle of the connection monitor task.
47    /// Aborted on disconnect/reconnect to prevent orphaned tasks.
48    pub monitor_handle: Mutex<Option<JoinHandle<()>>>,
49    /// LED write throttle state for coalescing rapid `set_leds` calls.
50    pub led_throttle: Mutex<LedThrottleState>,
51    /// JoinHandle of the LED debounce task.
52    pub led_debounce_handle: Mutex<Option<JoinHandle<()>>>,
53    /// Notify the LED debounce task that a new color is pending.
54    pub led_notify: Arc<Notify>,
55    /// Software calibration offset applied to accelerometer readings
56    /// before face value interpretation. `None` when no software
57    /// calibration has been performed.
58    ///
59    /// Uses `std::sync::RwLock` (not `tokio::sync::RwLock`) because the
60    /// lock is only held for a trivial copy — never across `.await`.
61    /// This avoids unnecessary task-scheduling overhead on every
62    /// sensor event in the notification task.
63    /// `Arc`-wrapped so it can be cloned into the notification task.
64    pub calibration_offset: Arc<StdRwLock<Option<AccelerationOffset>>>,
65    /// Last known charging state, updated by the notification task.
66    /// Stored as `AtomicU8` using `ChargingState` repr.
67    /// `Arc`-wrapped so it can be cloned into the notification task.
68    pub charging_state: Arc<AtomicU8>,
69}
