1use std::str::FromStr;
2
3use dice_rs::model::led::LedColor;
4use dice_rs::model::led::PulseBlinkMode;
5use dice_rs::model::led::PulseLeds;
6use dice_rs::service::manager::DiceManager;
7
8use crate::cli_error::CliError;
9use crate::cli_error::Result;
10use crate::led_action::LedAction;
11
12pub async fn run(manager: &DiceManager, address: &str, action: LedAction) -> Result<()> {
13    let dice = manager.connect_by_address(address).await?;
14
15    match action {
16        LedAction::Set { color } => {
17            let color = LedColor::from_str(&color)?;
18            dice.set_leds_immediate(color, color).await?;
19            println!("LEDs set to {color}");
20        }
21        LedAction::SetDual { led1, led2 } => {
22            let led1 = LedColor::from_str(&led1)?;
23            let led2 = LedColor::from_str(&led2)?;
24            dice.set_leds_immediate(led1, led2).await?;
25            println!("LED 1: {led1}, LED 2: {led2}");
26        }
27        LedAction::Pulse {
28            color,
29            count,
30            on_time,
31            off_time,
32            blink_mode,
33            leds,
34        } => {
35            let color = LedColor::from_str(&color)?;
36            let blink_mode: PulseBlinkMode = blink_mode.parse().map_err(CliError::from)?;
37            let leds: PulseLeds = leds.parse().map_err(CliError::from)?;
38            dice.pulse_leds(count, on_time, off_time, color, blink_mode, leds).await?;
39            println!("Pulsing {color} x{count} ({on_time}0ms on / {off_time}0ms off)");
40        }
41        LedAction::Off => {
42            dice.set_leds_immediate(LedColor::OFF, LedColor::OFF).await?;
43            println!("LEDs off");
44        }
45    }
46
47    dice.disconnect().await?;
48    Ok(())
49}
