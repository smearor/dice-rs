15use crate::cli_error::Result;
16use crate::command::Command;
17use crate::output_format::OutputFormat;
18use dice_rs::service::manager::DiceManager;

20/// Run a subcommand.
21pub async fn run_command(manager: &DiceManager, command: Command, format: OutputFormat) -> Result<()> {
22    match command {
23        Command::Scan { duration } => scan::run(manager, duration, format).await,
24        Command::Listen { address, dice_type } => listen::run(manager, &address, &dice_type, format).await,
25        Command::Battery { address } => battery::run(manager, &address, format).await,
26        Command::Led { address, action } => led::run(manager, &address, action).await,
27        Command::Tap { address, enable } => tap::run(manager, &address, enable).await,
28        Command::DoubleTap { address, enable } => tap::run_double(manager, &address, enable).await,
29        Command::Calibrate { address } => calibrate::run(manager, &address).await,
30        Command::Status { address } => status::run(manager, &address, format).await,
31        Command::Color { address } => color::run(manager, &address, format).await,
32        Command::Charging { address } => charging::run(manager, &address, format).await,
33        Command::Disconnect { address } => disconnect::run(manager, &address).await,
34        Command::DisconnectAll => disconnect_all::run(manager).await,
35        Command::Interactive => interactive::run(manager).await,
36    }
37}
