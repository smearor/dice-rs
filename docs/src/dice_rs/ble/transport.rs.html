19/// Abstraction over a BLE backend (btleplug, bluer, mock).
20#[async_trait]
21pub trait BleTransport: Send + Sync {
22    /// The peripheral type returned by the backend.
23    type Peripheral: BlePeripheral + Send + Sync;
24
25    /// Start scanning for BLE devices with the given filter.
26    async fn start_scan(&self, filter: ScanFilter) -> Result<()>;
27
28    /// Stop an active scan.
29    async fn stop_scan(&self) -> Result<()>;
30
31    /// Return all peripherals discovered so far.
32    async fn peripherals(&self) -> Result<Vec<Self::Peripheral>>;
33
34    /// Subscribe to central events (device discovered, lost, etc.).
35    async fn events(&self) -> Result<BoxStream<'static, btleplug::api::CentralEvent>>;
36}

38/// Abstraction over a single BLE peripheral.
39#[async_trait]
40pub trait BlePeripheral: Send + Sync {
41    /// Unique identifier of the peripheral.
42    fn id(&self) -> btleplug::platform::PeripheralId;
43
44    /// MAC address of the peripheral.
45    fn address(&self) -> btleplug::api::BDAddr;
46
47    /// Current connection state.
48    async fn is_connected(&self) -> Result<bool>;
49
50    /// Cached properties (local name, RSSI, etc.).
51    async fn properties(&self) -> Result<Option<btleplug::api::PeripheralProperties>>;
52
53    /// Read the current RSSI (signal strength) in dBm.
54    ///
55    /// On Linux returns the latest RSSI from BlueZ device properties.
56    /// May return `NotConnected` if no RSSI value is available.
57    async fn read_rssi(&self) -> Result<i16>;
58
59    /// Establish a connection.
60    async fn connect(&self) -> Result<()>;
61
62    /// Disconnect from the peripheral.
63    async fn disconnect(&self) -> Result<()>;
64
65    /// Discover GATT services and characteristics.
66    async fn discover_services(&self) -> Result<()>;
67
68    /// Find a characteristic by UUID.
69    fn characteristic(&self, uuid: Uuid) -> Option<btleplug::api::Characteristic>;
70
71    /// Write data to a characteristic.
72    async fn write(&self, characteristic: &btleplug::api::Characteristic, data: &[u8], write_type: WriteType) -> Result<()>;
73
74    /// Enable notifications on a characteristic.
75    async fn subscribe(&self, characteristic: &btleplug::api::Characteristic) -> Result<()>;
76
77    /// Stream of incoming notifications.
78    async fn notifications(&self) -> Result<BoxStream<'static, btleplug::api::ValueNotification>>;
79}

81/// btleplug implementation of `BleTransport`.
82///
83/// Wraps `btleplug::platform::Adapter` which provides `Central` functionality.
84pub struct BtleplugTransport {
85    adapter: Adapter,
86}
87
88impl BtleplugTransport {
89    /// Create a new transport by selecting the first available Bluetooth adapter.
90    pub async fn new() -> Result<Self> {
91        let manager = Manager::new().await.map_err(|e| DiceError::from(BleError::scan(e)))?;
92        let adapters = manager.adapters().await.map_err(|e| DiceError::from(BleError::scan(e)))?;
93        let adapter = adapters
94            .into_iter()
95            .next()
96            .ok_or_else(|| DiceError::from(BleError::scan("no Bluetooth adapter found")))?;
97        Ok(Self { adapter })
98    }
99
100    /// Returns the write characteristic UUID (NUS RX).
101    pub fn write_char_uuid() -> Uuid {
102        NUS_WRITE_CHAR_UUID
103    }
104
105    /// Returns the notify characteristic UUID (NUS TX).
106    pub fn notify_char_uuid() -> Uuid {
107        NUS_NOTIFY_CHAR_UUID
108    }
109}

111#[async_trait]
112impl BleTransport for BtleplugTransport {
113    type Peripheral = BtleplugPeripheralWrapper;
114
115    async fn start_scan(&self, filter: ScanFilter) -> Result<()> {
116        self.adapter.start_scan(filter).await.map_err(|e| BleError::scan(e).into())
117    }
118
119    async fn stop_scan(&self) -> Result<()> {
120        self.adapter.stop_scan().await.map_err(|e| BleError::scan(e).into())
121    }
122
123    async fn peripherals(&self) -> Result<Vec<Self::Peripheral>> {
124        let peripherals = self.adapter.peripherals().await.map_err(|e| DiceError::from(BleError::scan(e)))?;
125        Ok(peripherals.into_iter().map(BtleplugPeripheralWrapper::new).collect())
126    }
127
128    async fn events(&self) -> Result<BoxStream<'static, btleplug::api::CentralEvent>> {
129        let events = self.adapter.events().await.map_err(|e| DiceError::from(BleError::scan(e)))?;
130        Ok(Box::pin(events))
131    }
132}

134/// Wrapper around `btleplug::platform::Peripheral` implementing `BlePeripheral`.
135#[derive(Clone)]
136pub struct BtleplugPeripheralWrapper {
137    inner: BtleplugPeripheral,
138}
139
140impl BtleplugPeripheralWrapper {
141    /// Create a new wrapper from a btleplug peripheral.
142    pub fn new(peripheral: BtleplugPeripheral) -> Self {
143        Self { inner: peripheral }
144    }
145
146    /// Returns a reference to the inner btleplug peripheral.
147    pub fn inner(&self) -> &BtleplugPeripheral {
148        &self.inner
149    }
150}
151
152#[async_trait]
153impl BlePeripheral for BtleplugPeripheralWrapper {
154    fn id(&self) -> btleplug::platform::PeripheralId {
155        self.inner.id()
156    }
157
158    fn address(&self) -> btleplug::api::BDAddr {
159        self.inner.address()
160    }
161
162    async fn is_connected(&self) -> Result<bool> {
163        self.inner.is_connected().await.map_err(|e| BleError::not_connected(e).into())
164    }
165
166    async fn properties(&self) -> Result<Option<btleplug::api::PeripheralProperties>> {
167        self.inner.properties().await.map_err(|e| BleError::not_connected(e).into())
168    }
169
170    async fn read_rssi(&self) -> Result<i16> {
171        self.inner.read_rssi().await.map_err(|e| BleError::not_connected(e).into())
172    }
173
174    async fn connect(&self) -> Result<()> {
175        self.inner.connect().await.map_err(|e| BleError::connect(e).into())
176    }
177
178    async fn disconnect(&self) -> Result<()> {
179        self.inner.disconnect().await.map_err(|e| BleError::disconnect(e).into())
180    }
181
182    async fn discover_services(&self) -> Result<()> {
183        self.inner.discover_services().await.map_err(|e| BleError::discovery(e).into())
184    }
185
186    fn characteristic(&self, uuid: Uuid) -> Option<btleplug::api::Characteristic> {
187        self.inner.characteristics().iter().find(|c| c.uuid == uuid).cloned()
188    }
189
190    async fn write(&self, characteristic: &btleplug::api::Characteristic, data: &[u8], write_type: WriteType) -> Result<()> {
191        self.inner.write(characteristic, data, write_type).await.map_err(|e| BleError::write(e).into())
192    }
193
194    async fn subscribe(&self, characteristic: &btleplug::api::Characteristic) -> Result<()> {
195        self.inner.subscribe(characteristic).await.map_err(|e| BleError::subscribe(e).into())
196    }
197
198    async fn notifications(&self) -> Result<BoxStream<'static, btleplug::api::ValueNotification>> {
199        let notifications = self.inner.notifications().await.map_err(|e| DiceError::from(BleError::subscribe(e)))?;
200        Ok(Box::pin(notifications))
201    }
