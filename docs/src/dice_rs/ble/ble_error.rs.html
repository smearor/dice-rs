3/// Errors originating from the BLE transport layer.
4///
5/// Each variant carries the underlying backend error message as a `String`
6/// for diagnostics, without exposing the backend (btleplug) error type to
7/// consumers of this crate.
8#[derive(Debug, Clone, PartialEq, thiserror::Error)]
9pub enum BleError {
10    /// BLE scan or adapter operation failed (start, stop, enumerate, events).
11    #[error("BLE scan failed: {0}")]
12    Scan(String),
13
14    /// BLE connection attempt failed (backend error).
15    #[error("connection failed: {0}")]
16    Connect(String),
17
18    /// No connection attempt was made yet.
19    #[error("no connection attempt made")]
20    NoAttemptMade,
21
22    /// Device with the given address was not found in scan results.
23    #[error("device not found: {address}")]
24    DeviceNotFound { address: String },
25
26    /// Peripheral was not found among discovered BLE peripherals.
27    #[error("peripheral not found for {name}")]
28    PeripheralNotFound { name: String },
29
30    /// BLE disconnect operation failed.
31    #[error("disconnect failed: {0}")]
32    Disconnect(String),
33
34    /// BLE connection state check or property query failed while not connected.
35    #[error("not connected: {0}")]
36    NotConnected(String),
37
38    /// GATT service discovery failed.
39    #[error("service discovery failed: {0}")]
40    Discovery(String),
41
42    /// A required GATT characteristic was not found.
43    #[error("characteristic not found: {0}")]
44    CharacteristicNotFound(NusCharacteristic),
45
46    /// GATT characteristic write failed.
47    #[error("write failed: {0}")]
48    Write(String),
49
50    /// GATT subscribe or notify operation failed.
51    #[error("subscribe failed: {0}")]
52    Subscribe(String),
53
54    /// Connection was lost during an operation.
55    #[error("connection lost")]
56    ConnectionLost,
57
58    /// Reconnect attempts exhausted without success.
59    #[error("reconnect failed after max retries")]
60    ReconnectFailed,
61}
62
63impl BleError {
64    /// Create a `Scan` error from any displayable error.
65    pub fn scan(e: impl std::fmt::Display) -> Self {
66        Self::Scan(e.to_string())
67    }
68
69    /// Create a `Connect` error from any displayable error.
70    pub fn connect(e: impl std::fmt::Display) -> Self {
71        Self::Connect(e.to_string())
72    }
73
74    /// Create a `DeviceNotFound` error for the given address.
75    pub fn device_not_found(address: impl std::fmt::Display) -> Self {
76        Self::DeviceNotFound { address: address.to_string() }
77    }
78
79    /// Create a `PeripheralNotFound` error for the given device name.
80    pub fn peripheral_not_found(name: impl std::fmt::Display) -> Self {
81        Self::PeripheralNotFound { name: name.to_string() }
82    }
83
84    /// Create a `Disconnect` error from any displayable error.
85    pub fn disconnect(e: impl std::fmt::Display) -> Self {
86        Self::Disconnect(e.to_string())
87    }
88
89    /// Create a `NotConnected` error from any displayable error.
90    pub fn not_connected(e: impl std::fmt::Display) -> Self {
91        Self::NotConnected(e.to_string())
92    }
93
94    /// Create a `Discovery` error from any displayable error.
95    pub fn discovery(e: impl std::fmt::Display) -> Self {
96        Self::Discovery(e.to_string())
97    }
98
99    /// Create a `Write` error from any displayable error.
100    pub fn write(e: impl std::fmt::Display) -> Self {
101        Self::Write(e.to_string())
102    }
103
104    /// Create a `Subscribe` error from any displayable error.
105    pub fn subscribe(e: impl std::fmt::Display) -> Self {
106        Self::Subscribe(e.to_string())
107    }
108
109    /// Create a `CharacteristicNotFound` error for the given NUS characteristic.
110    pub fn characteristic_not_found(char: NusCharacteristic) -> Self {
111        Self::CharacteristicNotFound(char)
112    }
113}
