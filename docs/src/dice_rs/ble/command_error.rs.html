1/// Errors that can occur when encoding or decoding commands.
2#[derive(Debug, Clone, PartialEq, thiserror::Error)]
3pub enum CommandError {
4    /// The packet is empty.
5    #[error("empty packet")]
6    EmptyPacket,
7    /// The opcode is not a known command.
8    #[error("unknown opcode: 0x{opcode:02X} (length {length})")]
9    UnknownOpcode { opcode: u8, length: usize },
10    /// The packet length does not match the expected payload size.
11    #[error("invalid payload length: expected {expected}, got {actual}")]
12    InvalidLength { expected: usize, actual: usize },
13}
