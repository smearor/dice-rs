1/// Errors that can occur when parsing a notification packet.
2#[derive(Debug, Clone, PartialEq, Eq, thiserror::Error)]
3pub enum ParseError {
4    /// The packet is empty.
5    #[error("empty packet")]
6    EmptyPacket,
7    /// The packet is shorter than expected.
8    #[error("truncated packet: expected {expected} bytes, got {actual}")]
9    TruncatedPacket { expected: usize, actual: usize },
10    /// The first byte does not match any known event.
11    #[error("unknown event byte: 0x{byte:02X}")]
12    UnknownEvent { byte: u8 },
13    /// The color byte is not a valid DiceColor value.
14    #[error("invalid dice color value: {0}")]
15    InvalidColor(u8),
16}
