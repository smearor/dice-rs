5/// Errors that can occur when interacting with a GoDice.
6#[derive(Debug, Clone, PartialEq, thiserror::Error)]
7pub enum DiceError {
8    /// BLE transport error (scan, connect, disconnect, GATT operations).
9    #[error(transparent)]
10    Ble(#[from] BleError),
11    /// A mutex lock was poisoned by a panicking thread.
12    #[error("lock poisoned")]
13    LockPoisoned,
14    /// A request-response query timed out before the dice responded.
15    #[error("response timeout: no reply within {0:?}")]
16    ResponseTimeout(Duration),
17    /// An invalid face value (0) was encountered.
18    #[error("invalid face value: {0}")]
19    InvalidFaceValue(u8),
20    /// An invalid dice type byte was encountered (e.g. from AtomicU8).
21    #[error("invalid dice type byte: {0}")]
22    InvalidDiceType(u8),
23    /// Calibration command failed. The dice reported a calibration error.
24    #[error("calibration failed")]
25    CalibrationFailed,
26    /// The calibration protocol is not yet implemented.
27    #[error("calibration protocol not yet confirmed")]
28    CalibrationNotConfirmed,
29    /// A notification packet could not be parsed.
30    #[error("parse error: {0}")]
31    Parse(#[from] crate::ble::parse_error::ParseError),
32    /// A command could not be encoded or decoded.
33    #[error("command error: {0}")]
34    Command(#[from] CommandError),
35    /// An invalid dice color byte was encountered.
36    #[error("invalid dice color: {0}")]
37    InvalidColor(u8),
38}
39
40/// Convenience type alias used throughout the crate.
41pub type Result<T> = std::result::Result<T, DiceError>;
