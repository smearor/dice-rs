19/// Trait for formatting values in the selected output format.
20///
21/// Implementations provide format-specific output (table, JSON, or plain text)
22/// for CLI display.
23///
24/// JSON output is derived automatically via `serde_json::to_string`.
25/// Plain text output defaults to `Display` formatting.
26/// Table output defaults to rendering `get_table_rows` via `tabled`.
27/// Implementors only need to provide `get_table_rows` and set `Row`.
28/// Types with non-standard table formatting (e.g. `DiceEvent`) override `format_table`.
29pub trait OutputFormatter: Display + Serialize {
30    /// Row type used for table rendering.
31    type Row: Tabled;
32
33    /// Build the rows for table rendering.
34    fn get_table_rows(&self) -> Result<Vec<Self::Row>>;
35
36    /// Format the value as a table.
37    ///
38    /// Defaults to rendering `get_table_rows` via `tabled` with rounded style.
39    fn format_table(&self) -> Result<String> {
40        Ok(tabled::Table::new(self.get_table_rows()?).with(tabled::settings::Style::rounded()).to_string())
41    }
42
43    /// Format the value as plain text.
44    ///
45    /// Defaults to `Display` formatting.
46    fn format_plain(&self) -> Result<String> {
47        Ok(format!("{self}"))
48    }
49
50    /// Format the value for the given output format.
51    fn format_output(&self, format: OutputFormat) -> Result<String> {
52        match format {
53            OutputFormat::Table => self.format_table(),
54            OutputFormat::Json => serde_json::to_string(self).map_err(CliError::JsonSerialize),
55            OutputFormat::Plain => self.format_plain(),
56        }
57    }
58}
59
60/// Print a value using its `OutputFormatter` implementation.
61pub fn print<T: OutputFormatter + ?Sized>(value: &T, format: OutputFormat) {
62    match value.format_output(format) {
63        Ok(msg) => println!("{msg}"),
64        Err(e) => eprintln!("{e}"),
65    }
