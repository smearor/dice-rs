1use crate::i18n;
2use crate::models::category_section::CategorySection;
3use crate::models::scorecard::Scorecard;
4use gtk4::prelude::*;
5
6/// Widget displaying the upper section bonus progress.
7///
8/// Shows the current upper section subtotal, the progress toward
9/// the 63-point bonus threshold, and whether the bonus has been achieved.
10pub struct BonusTracker {
11    /// The root container widget.
12    container: gtk4::Box,
13    /// Label showing the current subtotal.
14    subtotal_label: gtk4::Label,
15    /// The progress bar showing progress toward 63 points.
16    progress_bar: gtk4::ProgressBar,
17    /// Label showing the bonus status.
18    bonus_label: gtk4::Label,
19}
20
21impl BonusTracker {
22    /// Create a new bonus tracker.
23    pub fn new() -> Self {
24        let title = gtk4::Label::builder()
25            .label(&i18n::get("bonus-title"))
26            .css_classes(vec!["bonus-tracker-title"])
27            .halign(gtk4::Align::Start)
28            .build();
29
30        let subtotal_label = gtk4::Label::builder()
31            .css_classes(vec!["bonus-subtotal"])
32            .halign(gtk4::Align::Start)
33            .hexpand(true)
34            .build();
35
36        let progress_bar = gtk4::ProgressBar::builder()
37            .css_classes(vec!["bonus-progress"])
38            .halign(gtk4::Align::Fill)
39            .hexpand(true)
40            .build();
41
42        let bonus_label = gtk4::Label::builder()
43            .css_classes(vec!["bonus-status"])
44            .halign(gtk4::Align::Start)
45            .hexpand(true)
46            .build();
47
48        let container = gtk4::Box::builder()
49            .orientation(gtk4::Orientation::Vertical)
50            .css_classes(vec!["bonus-tracker"])
51            .spacing(4)
52            .halign(gtk4::Align::Fill)
53            .hexpand(true)
54            .build();
55
56        container.append(&title);
57        container.append(&subtotal_label);
58        container.append(&progress_bar);
59        container.append(&bonus_label);
60
61        Self {
62            container,
63            subtotal_label,
64            progress_bar,
65            bonus_label,
66        }
67    }
68
69    /// Update the bonus tracker with the current scorecard.
70    pub fn update(&self, scorecard: &Scorecard) {
71        let subtotal = scorecard.upper_subtotal();
72        let remaining = scorecard.upper_bonus_remaining();
73        let bonus = scorecard.upper_bonus();
74
75        let subtotal_text = i18n::get_int_int("bonus-subtotal", "current", subtotal.get() as i64, "threshold", CategorySection::UPPER_BONUS_THRESHOLD as i64);
76        self.subtotal_label.set_label(&subtotal_text);
77
78        let fraction = if subtotal.get() >= CategorySection::UPPER_BONUS_THRESHOLD {
79            1.0
80        } else {
81            subtotal.get() as f64 / CategorySection::UPPER_BONUS_THRESHOLD as f64
82        };
83        self.progress_bar.set_fraction(fraction);
84
85        let bonus_text = if bonus.get() > 0 {
86            i18n::get_int("bonus-achieved", "points", bonus.get() as i64)
87        } else if remaining.get() == 0 {
88            i18n::get("bonus-reached")
89        } else {
90            i18n::get_int("bonus-remaining", "remaining", remaining.get() as i64)
91        };
92        self.bonus_label.set_label(&bonus_text);
93
94        if bonus.get() > 0 {
95            self.bonus_label.add_css_class("bonus-achieved");
96        } else {
97            self.bonus_label.remove_css_class("bonus-achieved");
98        }
99    }
100
101    /// Clear the bonus tracker.
102    pub fn clear(&self) {
103        self.subtotal_label.set_label("");
104        self.progress_bar.set_fraction(0.0);
105        self.bonus_label.set_label("");
106        self.bonus_label.remove_css_class("bonus-achieved");
107    }
108
109    /// Get the root widget.
110    pub fn widget(&self) -> &gtk4::Widget {
111        self.container.upcast_ref()
112    }
113}
114
115impl Default for BonusTracker {
116    fn default() -> Self {
117        Self::new()
118    }
119}
