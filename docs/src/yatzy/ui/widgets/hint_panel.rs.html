11/// Widget displaying strategy hints for the current player.
12///
13/// Shows the recommended category to enter and the recommended
14/// dice to hold for re-rolling. Designed to help human players
15/// learn optimal strategy.
16pub struct HintPanel {
17    /// The root container widget.
18    container: gtk4::Box,
19    /// Label showing the recommended category.
20    category_label: gtk4::Label,
21    /// Label showing the expected score.
22    score_label: gtk4::Label,
23    /// Label showing the recommended hold dice.
24    hold_label: gtk4::Label,
25    /// The computer AI engine for generating hints.
26    ai: ComputerAi,
27}
28
29impl HintPanel {
30    /// Create a new hint panel.
31    pub fn new() -> Self {
32        let title = gtk4::Label::builder()
33            .label(&i18n::get("hint-title"))
34            .css_classes(vec!["hint-title"])
35            .halign(gtk4::Align::Start)
36            .build();
37
38        let category_label = gtk4::Label::builder()
39            .css_classes(vec!["hint-category"])
40            .halign(gtk4::Align::Start)
41            .hexpand(true)
42            .build();
43
44        let score_label = gtk4::Label::builder().css_classes(vec!["hint-score"]).halign(gtk4::Align::End).build();
45
46        let hold_label = gtk4::Label::builder()
47            .css_classes(vec!["hint-holds"])
48            .halign(gtk4::Align::Start)
49            .hexpand(true)
50            .build();
51
52        let category_row = gtk4::Box::builder().orientation(gtk4::Orientation::Horizontal).spacing(8).hexpand(true).build();
53        category_row.append(&category_label);
54        category_row.append(&score_label);
55
56        let container = gtk4::Box::builder()
57            .orientation(gtk4::Orientation::Vertical)
58            .css_classes(vec!["hint-panel"])
59            .spacing(4)
60            .halign(gtk4::Align::Fill)
61            .hexpand(true)
62            .build();
63
64        container.append(&title);
65        container.append(&category_row);
66        container.append(&hold_label);
67
68        Self {
69            container,
70            category_label,
71            score_label,
72            hold_label,
73            ai: ComputerAi::new(),
74        }
75    }
76
77    /// Update the hint panel with the current game state.
78    pub fn update(&self, dice: &DiceSet, scorecard: &Scorecard, rolls_used: RollCount) {
79        let best = self.ai.hint_best_category(dice, scorecard);
80        self.show_category_hint(&best);
81
82        if !rolls_used.is_exhausted() {
83            match self.ai.hint_best_holds(dice, scorecard, rolls_used) {
84                Ok(holds) => self.show_hold_hint(dice, holds),
85                Err(_) => self.hold_label.set_label("—"),
86            }
87        } else {
88            self.hold_label.set_label(&i18n::get("hint-no-more-rolls"));
89        }
90    }
91
92    /// Show the category recommendation.
93    fn show_category_hint(&self, choice: &CategoryChoice) {
94        let text = i18n::get_str("hint-category", "category", &choice.category().to_string());
95        self.category_label.set_label(&text);
96
97        let score_text = if choice.score().get() > 0 {
98            i18n::get_int("hint-score", "score", choice.score().get() as i64)
99        } else {
100            i18n::get("hint-score-zero")
101        };
102        self.score_label.set_label(&score_text);
103    }
104
105    /// Show the hold recommendation.
106    fn show_hold_hint(&self, dice: &DiceSet, holds: HoldMask) {
107        if holds.held_count() == 5 {
108            self.hold_label.set_label(&i18n::get("hint-hold-all"));
109            return;
110        }
111
112        if holds.held_count() == 0 {
113            self.hold_label.set_label(&i18n::get("hint-hold-none"));
114            return;
115        }
116
117        let held_values: Vec<String> = DiceSlot::all()
118            .filter(|slot| holds.is_held(*slot))
119            .map(|slot| dice.values()[slot.get() as usize].to_string())
120            .collect();
121
122        let text = i18n::get_str("hint-hold-some", "values", &held_values.join(", "));
123        self.hold_label.set_label(&text);
124    }
125
126    /// Clear the hint panel.
127    pub fn clear(&self) {
128        self.category_label.set_label("");
129        self.score_label.set_label("");
130        self.hold_label.set_label("");
131    }
132
133    /// Get the root widget.
134    pub fn widget(&self) -> &gtk4::Widget {
135        self.container.upcast_ref()
136    }
137}
138
139impl Default for HintPanel {
140    fn default() -> Self {
141        Self::new()
142    }
