1use crate::i18n;
2use crate::models::highscore::HighscoreList;
3use gtk4::prelude::*;
4
5/// Dialog displaying the highscore list.
6///
7/// Shows a ranked list of the top scores achieved across all game
8/// sessions. The list is loaded from disk via `HighscoreStore`.
9pub struct HighscoreDialog {
10    /// The GTK window.
11    dialog: gtk4::Window,
12}
13
14impl HighscoreDialog {
15    /// Create a new highscore dialog.
16    ///
17    /// If `list` contains entries, they are displayed in a ranked table.
18    /// If the list is empty, a placeholder message is shown instead.
19    pub fn new(parent: &gtk4::ApplicationWindow, list: &HighscoreList) -> Self {
20        let dialog = gtk4::Window::builder()
21            .title(&i18n::get("highscore-dialog-title"))
22            .modal(true)
23            .transient_for(parent)
24            .default_width(400)
25            .default_height(500)
26            .build();
27
28        let content = gtk4::Box::builder()
29            .orientation(gtk4::Orientation::Vertical)
30            .spacing(16)
31            .margin_start(24)
32            .margin_end(24)
33            .margin_top(24)
34            .margin_bottom(24)
35            .build();
36
37        let title = gtk4::Label::builder()
38            .label(&i18n::get("highscore-title"))
39            .css_classes(vec!["highscore-dialog-title"])
40            .halign(gtk4::Align::Center)
41            .build();
42        content.append(&title);
43
44        if list.is_empty() {
45            let empty_label = gtk4::Label::builder()
46                .label(&i18n::get("highscore-empty"))
47                .css_classes(vec!["dim-label", "highscore-empty"])
48                .halign(gtk4::Align::Center)
49                .valign(gtk4::Align::Center)
50                .vexpand(true)
51                .build();
52            content.append(&empty_label);
53        } else {
54            let scrolled = gtk4::ScrolledWindow::builder().hexpand(true).vexpand(true).build();
55
56            let standings_list = gtk4::ListBox::builder()
57                .css_classes(vec!["highscore-list"])
58                .halign(gtk4::Align::Center)
59                .hexpand(true)
60                .build();
61
62            for (index, entry) in list.entries().iter().enumerate() {
63                let row = gtk4::Box::builder()
64                    .orientation(gtk4::Orientation::Horizontal)
65                    .spacing(16)
66                    .css_classes(vec!["highscore-row"])
67                    .halign(gtk4::Align::Center)
68                    .build();
69
70                let rank_label = gtk4::Label::builder()
71                    .label(format!("{}.", index + 1))
72                    .css_classes(vec!["highscore-rank"])
73                    .build();
74
75                let name_label = gtk4::Label::builder()
76                    .label(entry.player_name())
77                    .css_classes(vec!["highscore-name"])
78                    .hexpand(true)
79                    .halign(gtk4::Align::Start)
80                    .build();
81
82                let score_label = gtk4::Label::builder()
83                    .label(i18n::get_int("highscore-score", "score", entry.score().get() as i64))
84                    .css_classes(vec!["highscore-score"])
85                    .build();
86
87                row.append(&rank_label);
88                row.append(&name_label);
89                row.append(&score_label);
90
91                let list_row = gtk4::ListBoxRow::builder().child(&row).build();
92                standings_list.append(&list_row);
93            }
94
95            scrolled.set_child(Some(&standings_list));
96            content.append(&scrolled);
97        }
98
99        let close_button = gtk4::Button::builder()
100            .label(&i18n::get("highscore-close"))
101            .css_classes(vec!["suggested-action", "highscore-close-button"])
102            .halign(gtk4::Align::Center)
103            .build();
104        content.append(&close_button);
105
106        {
107            let dialog_clone = dialog.clone();
108            close_button.connect_clicked(move |_| {
109                dialog_clone.close();
110            });
111        }
112
113        dialog.set_child(Some(&content));
114
115        Self { dialog }
116    }
117
118    /// Present the dialog.
119    pub fn present(&self) {
120        self.dialog.present();
121    }
122}
