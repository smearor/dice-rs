12/// Callback type for when a category is selected for scoring.
13type CategorySelectedCallback = Rc<dyn Fn(ScoreCategory)>;

15/// Widget displaying all players' scorecards as a grid.
16///
17/// Shows all 13 categories with scores for each player, organized into
18/// upper and lower sections. Empty categories in the active player's
19/// column are clickable for score entry. Filled categories show their
20/// score; crossed-out categories show a strike-through.
21pub struct ScorecardView {
22    /// The root container widget.
23    container: gtk4::Box,
24    /// The grid holding the scorecard rows.
25    grid: gtk4::Grid,
26    /// The currently displayed players (owned copies).
27    players: Rc<RefCell<Vec<Player>>>,
28    /// The index of the active player whose column is clickable.
29    active_player: Rc<RefCell<usize>>,
30    /// The last-entered (player, category) to highlight until next entry.
31    last_entered: Rc<RefCell<Option<(usize, ScoreCategory)>>>,
32    /// Callback invoked when a category is clicked.
33    on_category_selected: Rc<RefCell<Option<CategorySelectedCallback>>>,
34    /// CSS provider for dynamic player column colors.
35    css_provider: gtk4::CssProvider,
36}

38impl ScorecardView {
39    /// Create a new empty scorecard view.
40    pub fn new() -> Self {
41        let grid = gtk4::Grid::builder()
42            .css_classes(vec!["scorecard-grid"])
43            .column_spacing(8)
44            .row_spacing(2)
45            .vexpand(true)
46            .hexpand(true)
47            .build();
48
49        let container = gtk4::Box::builder()
50            .orientation(gtk4::Orientation::Vertical)
51            .css_classes(vec!["scorecard-view"])
52            .vexpand(true)
53            .hexpand(true)
54            .build();
55
56        container.append(&grid);
57
58        let css_provider = gtk4::CssProvider::new();
59        gtk4::style_context_add_provider_for_display(&gtk4::gdk::Display::default().unwrap(), &css_provider, gtk4::STYLE_PROVIDER_PRIORITY_APPLICATION);
60
61        Self {
62            container,
63            grid,
64            players: Rc::new(RefCell::new(Vec::new())),
65            active_player: Rc::new(RefCell::new(0)),
66            last_entered: Rc::new(RefCell::new(None)),
67            on_category_selected: Rc::new(RefCell::new(None)),
68            css_provider,
69        }
70    }

72    /// Set the callback invoked when a category is clicked.
73    pub fn connect_category_selected<F>(&self, callback: F)
74    where
75        F: Fn(ScoreCategory) + 'static,
76    {
77        *self.on_category_selected.borrow_mut() = Some(Rc::new(callback));

80    /// Update the view to display all players' scorecards.
81    pub fn update_players(&self, players: &[Player], active_player: usize) {
82        *self.players.borrow_mut() = players.to_vec();
83        *self.active_player.borrow_mut() = active_player;
84        self.rebuild_grid(players, active_player);
85    }

87    /// Mark a cell as recently entered so it stays highlighted until the next entry.
88    pub fn set_last_entered(&self, player_index: usize, category: ScoreCategory) {
89        *self.last_entered.borrow_mut() = Some((player_index, category));
90        let players = self.players.borrow().clone();
91        let active = *self.active_player.borrow();
92        self.rebuild_grid(&players, active);
93    }
94
95    /// Clear the recently-entered highlight.
96    pub fn clear_last_entered(&self) {
97        *self.last_entered.borrow_mut() = None;
98    }

100    /// Update the view to display a single player's scorecard.
101    /// Kept for backward compatibility — updates the active player's column.
102    pub fn update(&self, scorecard: &Scorecard) {
103        let mut players = self.players.borrow_mut();
104        let active = *self.active_player.borrow();
105        if active < players.len() {
106            players[active].scorecard_mut().clone_from(scorecard);
107        }
108        let players_clone = players.clone();
109        drop(players);
110        self.rebuild_grid(&players_clone, active);
111    }

113    /// Clear the scorecard view.
114    pub fn clear(&self) {
115        *self.players.borrow_mut() = Vec::new();
116        *self.active_player.borrow_mut() = 0;
117        *self.last_entered.borrow_mut() = None;
118        while let Some(child) = self.grid.first_child() {
119            self.grid.remove(&child);
120        }
121    }

123    /// Number of columns: 1 for category labels + 1 per player.
124    fn num_columns(&self, players: &[Player]) -> i32 {
125        1 + players.len() as i32
126    }
127
128    /// Rebuild the grid from all players' scorecards.
129    fn rebuild_grid(&self, players: &[Player], active_player: usize) {
130        // Remove all existing children
131        while let Some(child) = self.grid.first_child() {
132            self.grid.remove(&child);
133        }
134
135        // Generate dynamic CSS for player column colors
136        self.update_player_color_css(players);
137
138        let num_cols = self.num_columns(players);
139        let mut row: i32 = 0;
140
141        // Player header row: name + grand total score
142        row = self.add_player_header_row(row, players, active_player);
143
144        // Upper section header
145        row = self.add_section_header(row, &i18n::get("scorecard-upper-section"), num_cols);
146
147        // Upper section categories
148        for category in ScoreCategory::ALL {
149            if category.section() == CategorySection::Upper {
150                row = self.add_category_row(row, players, active_player, category);
151            }
152        }
153
154        // Upper bonus
155        row = self.add_summary_row(row, &i18n::get("scorecard-bonus"), players, Scorecard::upper_bonus, num_cols, false);
156
157        // Upper subtotal
158        row = self.add_summary_row(row, &i18n::get("scorecard-subtotal"), players, Scorecard::upper_subtotal, num_cols, false);
159
160        // Lower section header
161        row = self.add_section_header(row, &i18n::get("scorecard-lower-section"), num_cols);
162
163        // Lower section categories
164        for category in ScoreCategory::ALL {
165            if category.section() == CategorySection::Lower {
166                row = self.add_category_row(row, players, active_player, category);
167            }
168        }
169
170        // Yatzy bonus
171        row = self.add_summary_row(row, &i18n::get("scorecard-yatzy-bonus"), players, Scorecard::yatzy_bonus, num_cols, false);
172
173        // Lower subtotal
174        row = self.add_summary_row(row, &i18n::get("scorecard-subtotal"), players, Scorecard::lower_subtotal, num_cols, false);
175
176        // Grand total
177        self.add_summary_row(row, &i18n::get("scorecard-total"), players, Scorecard::grand_total, num_cols, true);
178    }

180    /// Add the player header row with name and grand total score.
181    fn add_player_header_row(&self, row: i32, players: &[Player], active_player: usize) -> i32 {
182        // Empty cell above category labels
183        let spacer = gtk4::Label::builder().build();
184        self.grid.attach(&spacer, 0, row, 1, 1);
185
186        for (i, player) in players.iter().enumerate() {
187            let card = gtk4::Box::builder()
188                .orientation(gtk4::Orientation::Vertical)
189                .css_classes(vec!["player-card", &format!("player-color-{i}")])
190                .spacing(2)
191                .halign(gtk4::Align::Center)
192                .build();
193
194            let name_label = gtk4::Label::builder()
195                .label(player.name().as_str())
196                .css_classes(vec!["scorecard-player-name", &format!("player-color-{i}")])
197                .halign(gtk4::Align::Center)
198                .build();
199
200            let total = player.scorecard().grand_total().get();
201            let score_label = gtk4::Label::builder()
202                .label(&total.to_string())
203                .css_classes(vec!["player-score", &format!("player-color-{i}")])
204                .halign(gtk4::Align::Center)
205                .build();
206
207            if i == active_player {
208                name_label.add_css_class("active");
209                card.add_css_class("active");
210            }
211
212            card.append(&name_label);
213            card.append(&score_label);
214            self.grid.attach(&card, (i + 1) as i32, row, 1, 1);
215        }
216
217        row + 1
218    }

233    /// Add a category row with label and per-player scores.
234    fn add_category_row(&self, mut row: i32, players: &[Player], active_player: usize, category: ScoreCategory) -> i32 {
235        // Category label — left-aligned, no hexpand so player columns start right after
236        let category_label = gtk4::Label::builder()
237            .label(category.to_string())
238            .css_classes(vec!["scorecard-category"])
239            .halign(gtk4::Align::Start)
240            .build();
241
242        self.grid.attach(&category_label, 0, row, 1, 1);
243
244        let last_entered = self.last_entered.borrow();
245
246        // Per-player score cells
247        for (i, player) in players.iter().enumerate() {
248            let scorecard = player.scorecard();
249            let entry = scorecard.entry(category);
250            let is_filled = entry.is_used();
251            let is_active = i == active_player;
252            let is_clickable = is_active && !is_filled;
253            let is_recent = last_entered.is_some_and(|(p, c)| p == i && c == category);
254
255            let cell_box = gtk4::Box::builder()
256                .orientation(gtk4::Orientation::Horizontal)
257                .css_classes(vec!["scorecard-row", &format!("player-color-{i}")])
258                .spacing(4)
259                .halign(gtk4::Align::Center)
260                .hexpand(true)
261                .build();
262
263            if is_filled {
264                cell_box.add_css_class("filled");
265            }
266            if is_active {
267                cell_box.add_css_class("active");
268            }
269            if is_clickable {
270                cell_box.add_css_class("clickable");
271            }
272            if is_recent {
273                cell_box.add_css_class("recently-entered");
274            }
275
276            let score_text = match entry {
277                ScoreEntry::Empty => String::new(),
278                ScoreEntry::Filled(score) => score.get().to_string(),
279                ScoreEntry::CrossedOut => "—".to_string(),
280            };
281
282            let score_label = gtk4::Label::builder()
283                .label(&score_text)
284                .css_classes(vec!["scorecard-score"])
285                .halign(gtk4::Align::End)
286                .build();
287
288            if matches!(entry, ScoreEntry::CrossedOut) {
289                score_label.add_css_class("crossed-out");
290            }
291
292            cell_box.append(&score_label);
293
294            // Make clickable only for the active player's unfilled categories
295            if is_clickable {
296                let click = gtk4::GestureClick::new();
297                let category_for_click = category;
298                let on_selected = self.on_category_selected.clone();
299                click.connect_pressed(move |_, _, _, _| {
300                    if let Some(callback) = on_selected.borrow().as_ref() {
301                        callback(category_for_click);
302                    }
303                });
304                cell_box.add_controller(click);
305            }
306
307            self.grid.attach(&cell_box, (i + 1) as i32, row, 1, 1);
308        }
309
310        // Thin separator line spanning all columns
311        row += 1;
312        let num_cols = self.num_columns(players);
313        let separator = gtk4::Box::builder()
314            .css_classes(vec!["scorecard-row-separator"])
315            .hexpand(true)
316            .halign(gtk4::Align::Fill)
317            .build();
318        self.grid.attach(&separator, 0, row, num_cols, 1);
319
320        row + 1
321    }

323    /// Add a summary row (bonus, subtotal, total) with per-player values.
324    fn add_summary_row(&self, row: i32, label: &str, players: &[Player], score_fn: fn(&Scorecard) -> Score, num_cols: i32, is_grand_total: bool) -> i32 {
325        let mut css_classes = vec!["scorecard-total-row"];
326        if is_grand_total {
327            css_classes.push("scorecard-grand-total");
328        }
329
330        let row_box = gtk4::Box::builder()
331            .orientation(gtk4::Orientation::Horizontal)
332            .css_classes(css_classes)
333            .spacing(8)
334            .halign(gtk4::Align::Start)
335            .build();
336
337        let label_widget = gtk4::Label::builder().label(label).halign(gtk4::Align::Start).build();
338        row_box.append(&label_widget);
339        self.grid.attach(&row_box, 0, row, 1, 1);
340
341        for (i, player) in players.iter().enumerate() {
342            let score = score_fn(player.scorecard());
343
344            let score_widget = gtk4::Label::builder()
345                .label(score.get().to_string())
346                .css_classes(vec!["scorecard-score", &format!("player-color-{i}")])
347                .halign(gtk4::Align::Center)
348                .hexpand(true)
349                .build();
350
351            if i == *self.active_player.borrow() {
352                score_widget.add_css_class("active");
353            }
354
355            self.grid.attach(&score_widget, (i + 1) as i32, row, 1, 1);
356        }
357
358        let _ = num_cols;
359        row + 1
360    }

362    /// Generate dynamic CSS for player column colors.
363    ///
364    /// Each player gets a `player-color-N` CSS class that sets:
365    /// - The border color for score cells in their column
366    /// - The text color for their name label
367    ///
368    /// Colors are brightened by blending toward white to ensure
369    /// visibility in dark mode themes.
370    fn update_player_color_css(&self, players: &[Player]) {
371        let mut css = String::new();
372        for (i, player) in players.iter().enumerate() {
373            let color = player.color().led_color();
374            let brighten = |c: u8| -> u8 { ((c as f32 * 0.65) + (255.0 * 0.35)).round() as u8 };
375            let r = brighten(color.r);
376            let g = brighten(color.g);
377            let b = brighten(color.b);
378            let hex = format!("#{:02X}{:02X}{:02X}", r, g, b);
379            css.push_str(&format!(
380                ".player-color-{i} {{ border-color: {hex}; }}\n.scorecard-player-name.player-color-{i} {{ color: {hex}; }}\n"
381            ));
382        }
383        self.css_provider.load_from_data(&css);
384    }
