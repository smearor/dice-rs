12/// Service for controlling GoDice LED effects.
13///
14/// Translates logical `LedEffect`s into BLE commands sent to the
15/// physical dice. Uses `SlotMapping` to find the correct `Dice` handle
16/// for each slot.
17#[derive(Clone)]
18pub struct LedService {
19    /// Mapping from game slots to physical dice.
20    mapping: SlotMapping,
21}
22
23impl LedService {
24    /// Create a new LED service with the given slot mapping.
25    pub fn new(mapping: SlotMapping) -> Self {
26        Self { mapping }
27    }
28
29    /// Apply an LED effect to the physical dice.
30    ///
31    /// Returns `Ok(())` if the effect was applied to at least one die.
32    /// Errors from individual dice are logged and do not abort the
33    /// operation — partial LED failures are non-fatal.
34    pub async fn apply(&self, effect: &LedEffect) -> Result<()> {
35        match effect {
36            LedEffect::SolidAll { color } => {
37                self.set_all_leds(color.led_color()).await;
38            }
39            LedEffect::Solid { slot, color } => {
40                self.set_slot_led(*slot, color.led_color()).await;
41            }
42            LedEffect::OffAll => {
43                self.set_all_leds(LedColor::OFF).await;
44            }
45            LedEffect::Off { slot } => {
46                self.set_slot_led(*slot, LedColor::OFF).await;
47            }
48            LedEffect::HoldVisualization { holds } => {
49                self.apply_hold_visualization(holds, PlayerColor::GREEN).await;
50            }
51            LedEffect::HoldVisualizationWithColor { holds, held_color } => {
52                self.apply_hold_visualization(holds, *held_color).await;
53            }
54            LedEffect::Celebrate { pulse_count, color } => {
55                self.pulse_all(*pulse_count, color.led_color()).await;
56            }
57            LedEffect::Pulse { slot, pulse_count, color } => {
58                self.pulse_slot(*slot, *pulse_count, color.led_color()).await;
59            }
60            LedEffect::Rainbow { pulse_count } => {
61                self.rainbow_all(*pulse_count).await;
62            }
63        }
64        Ok(())
65    }
66
67    /// Set all connected dice to a solid color.
68    async fn set_all_leds(&self, color: LedColor) {
69        for (slot, dice) in self.mapping.assigned() {
70            if let Err(error) = dice.set_led(color).await {
71                debug!(slot = %slot.get(), error = %error, "failed to set LED on dice");
72            }
73        }
74    }
75
76    /// Set a single die's LED to a solid color.
77    async fn set_slot_led(&self, slot: DiceSlot, color: LedColor) {
78        if let Some(dice) = self.mapping.get(slot)
79            && let Err(error) = dice.set_led(color).await
80        {
81            debug!(slot = %slot.get(), error = %error, "failed to set LED on dice");
82        }
83    }
84
85    /// Apply hold visualization: held dice glow, others off.
86    async fn apply_hold_visualization(&self, holds: &HoldMask, held_color: PlayerColor) {
87        for slot in DiceSlot::all() {
88            let color = if holds.is_held(slot) { held_color.led_color() } else { LedColor::OFF };
89            if let Some(dice) = self.mapping.get(slot)
90                && let Err(error) = dice.set_led(color).await
91            {
92                debug!(slot = %slot.get(), error = %error, "failed to set LED for hold visualization");
93            }
94        }
95    }
96
97    /// Pulse all connected dice with a color.
98    async fn pulse_all(&self, _pulse_count: u8, color: LedColor) {
99        for (slot, dice) in self.mapping.assigned() {
100            if let Err(error) = dice.pulse_once(50, 50, color).await {
101                debug!(slot = %slot.get(), error = %error, "failed to pulse dice");
102            }
103        }
104    }
105
106    /// Pulse a single die with a color.
107    async fn pulse_slot(&self, slot: DiceSlot, pulse_count: u8, color: LedColor) {
108        if let Some(dice) = self.mapping.get(slot)
109            && let Err(error) = dice.pulse_once(pulse_count, 50, color).await
110        {
111            debug!(slot = %slot.get(), error = %error, "failed to pulse dice");
112        }
113    }
114
115    /// Rainbow pulse on all connected dice.
116    async fn rainbow_all(&self, pulse_count: u8) {
117        for (slot, dice) in self.mapping.assigned() {
118            if let Err(error) = dice
119                .pulse_leds(pulse_count, 50, 50, LedColor::WHITE, PulseBlinkMode::Rainbow, PulseLeds::Both)
120                .await
121            {
122                debug!(slot = %slot.get(), error = %error, "failed to rainbow pulse dice");
123            }
124        }
125    }
126}
