11/// A logical LED effect to apply to the physical dice.
12///
13/// Effects are translated to BLE commands by `LedService`.
14/// This abstraction allows the game logic to request effects
15/// without knowing BLE protocol details.
16#[derive(Debug, Clone, PartialEq, Eq)]
17pub enum LedEffect {
18    /// Set all dice to a solid color (e.g. active player color).
19    SolidAll {
20        /// The color to set on all dice.
21        color: PlayerColor,
22    },
23    /// Set a single die to a specific color.
24    Solid {
25        /// The slot of the die to update.
26        slot: DiceSlot,
27        /// The color to set.
28        color: PlayerColor,
29    },
30    /// Turn off all dice LEDs.
31    OffAll,
32    /// Turn off a single die's LEDs.
33    Off {
34        /// The slot of the die to turn off.
35        slot: DiceSlot,
36    },
37    /// Visualize held dice: held dice glow green, others are off.
38    HoldVisualization {
39        /// Which dice are held.
40        holds: HoldMask,
41    },
42    /// Visualize held dice with a custom color.
43    HoldVisualizationWithColor {
44        /// Which dice are held.
45        holds: HoldMask,
46        /// Color for held dice.
47        held_color: PlayerColor,
48    },
49    /// Pulse all dice with a celebration color (e.g. Yatzy, Full House).
50    Celebrate {
51        /// Number of pulses.
52        pulse_count: u8,
53        /// Color to pulse.
54        color: PlayerColor,
55    },
56    /// Pulse a single die.
57    Pulse {
58        /// The slot of the die to pulse.
59        slot: DiceSlot,
60        /// Number of pulses.
61        pulse_count: u8,
62        /// Color to pulse.
63        color: PlayerColor,
64    },
65    /// Rainbow pulse on all dice (for special events).
66    Rainbow {
67        /// Number of pulses.
68        pulse_count: u8,
69    },
70}
71
72impl LedEffect {
73    /// Celebration effect for rolling a Yatzy (5 of a kind).
74    pub fn yatzy() -> Self {
75        Self::Celebrate {
76            pulse_count: 5,
77            color: PlayerColor::GREEN,
78        }
79    }
80
81    /// Celebration effect for rolling a Full House.
82    pub fn full_house() -> Self {
83        Self::Celebrate {
84            pulse_count: 3,
85            color: PlayerColor::YELLOW,
86        }
87    }
88
89    /// Celebration effect for rolling a Large Straight.
90    pub fn large_straight() -> Self {
91        Self::Celebrate {
92            pulse_count: 3,
93            color: PlayerColor::CYAN,
94        }
95    }
96
97    /// Active player indicator: all dice glow in the player's color.
98    pub fn active_player(color: PlayerColor) -> Self {
99        Self::SolidAll { color }
100    }
101
102    /// Turn all dice off.
103    pub fn all_off() -> Self {
104        Self::OffAll
105    }
106
107    /// Convert a `PlayerColor` to `LedColor`.
108    #[cfg(test)]
109    pub(crate) fn led_color(color: PlayerColor) -> LedColor {
110        color.led_color()
111    }
112
113    /// Get the pulse parameters for this effect, if it is a pulse effect.
114    #[cfg(test)]
115    pub(crate) fn pulse_params(&self) -> Option<(u8, LedColor, PulseBlinkMode, PulseLeds)> {
116        match self {
117            Self::Celebrate { pulse_count, color } => Some((*pulse_count, Self::led_color(*color), PulseBlinkMode::Color, PulseLeds::Both)),
118            Self::Rainbow { pulse_count } => Some((*pulse_count, LedColor::WHITE, PulseBlinkMode::Rainbow, PulseLeds::Both)),
119            _ => None,
120        }
121    }
122}

125mod tests {

128    #[test]
129    fn yatzy_effect() {
130        let effect = LedEffect::yatzy();
131        assert_eq!(
132            effect,
133            LedEffect::Celebrate {
134                pulse_count: 5,
135                color: PlayerColor::GREEN
136            }
137        );
138    }
139
140    #[test]
141    fn full_house_effect() {
142        let effect = LedEffect::full_house();
143        assert_eq!(
144            effect,
145            LedEffect::Celebrate {
146                pulse_count: 3,
147                color: PlayerColor::YELLOW
148            }
149        );
150    }
151
152    #[test]
153    fn active_player_effect() {
154        let effect = LedEffect::active_player(PlayerColor::RED);
155        assert_eq!(effect, LedEffect::SolidAll { color: PlayerColor::RED });
156    }
157
158    #[test]
159    fn pulse_params_for_celebrate() {
160        let effect = LedEffect::yatzy();
161        let params = effect.pulse_params();
162        assert!(params.is_some());
163        let (count, color, mode, leds) = params.unwrap();
164        assert_eq!(count, 5);
165        assert_eq!(color, LedColor::GREEN);
166        assert_eq!(mode, PulseBlinkMode::Color);
167        assert_eq!(leds, PulseLeds::Both);
168    }
169
170    #[test]
171    fn pulse_params_for_rainbow() {
172        let effect = LedEffect::Rainbow { pulse_count: 3 };
173        let params = effect.pulse_params();
174        assert!(params.is_some());
175        let (_, _, mode, _) = params.unwrap();
176        assert_eq!(mode, PulseBlinkMode::Rainbow);
177    }
178
179    #[test]
180    fn pulse_params_none_for_solid() {
181        let effect = LedEffect::SolidAll { color: PlayerColor::RED };
182        assert!(effect.pulse_params().is_none());
183    }
184
185    #[test]
186    fn all_off() {
187        assert_eq!(LedEffect::all_off(), LedEffect::OffAll);
