10/// Persists game settings to a JSON file on disk.
11///
12/// The store handles loading and saving `GameSettings` so that
13/// the player setup (names, colors, mode) is restored between
14/// game sessions.
15pub struct SettingsStore {
16    /// The path to the settings file.
17    file_path: PathBuf,
18}
19
20impl SettingsStore {
21    /// Create a new settings store pointing to the given file path.
22    pub fn new(file_path: impl Into<PathBuf>) -> Self {
23        Self { file_path: file_path.into() }
24    }
25
26    /// Create a settings store using the default path in the user's config directory.
27    ///
28    /// Uses `XDG_CONFIG_HOME/yatzy/settings.json` on Linux.
29    pub fn default_path() -> Result<Self> {
30        let config_dir = dirs::config_dir().ok_or(YatzyError::SettingsIoError("no config directory".to_string()))?;
31        let settings_dir = config_dir.join("yatzy");
32        let file_path = settings_dir.join("settings.json");
33        Ok(Self { file_path })
34    }
35
36    /// Load settings from disk.
37    ///
38    /// Returns `Ok(None)` if the file does not exist (first run).
39    pub async fn load(&self) -> Result<Option<GameSettings>> {
40        if !self.file_path.exists() {
41            return Ok(None);
42        }
43        let mut file = fs::File::open(&self.file_path).await.map_err(|e| YatzyError::SettingsIoError(e.to_string()))?;
44        let mut contents = String::new();
45        file.read_to_string(&mut contents)
46            .await
47            .map_err(|e| YatzyError::SettingsIoError(e.to_string()))?;
48        let settings: GameSettings = serde_json::from_str(&contents).map_err(|e| YatzyError::SettingsParseError(e.to_string()))?;
49        Ok(Some(settings))
50    }
51
52    /// Save settings to disk.
53    ///
54    /// Creates parent directories if they don't exist.
55    pub async fn save(&self, settings: &GameSettings) -> Result<()> {
56        if let Some(parent) = self.file_path.parent() {
57            fs::create_dir_all(parent).await.map_err(|e| YatzyError::SettingsIoError(e.to_string()))?;
58        }
59        let json = serde_json::to_string_pretty(settings).map_err(|e| YatzyError::SettingsParseError(e.to_string()))?;
60        let mut file = fs::File::create(&self.file_path)
61            .await
62            .map_err(|e| YatzyError::SettingsIoError(e.to_string()))?;
63        file.write_all(json.as_bytes()).await.map_err(|e| YatzyError::SettingsIoError(e.to_string()))?;
64        Ok(())
65    }
66
67    /// Get the file path used by this store.
68    pub fn path(&self) -> &Path {
69        &self.file_path
70    }
71}

74mod tests {
75    use super::*;
76    use crate::models::game_mode::GameMode;
77    use crate::models::game_settings::PlayerSettingsEntry;
78    use crate::models::player_color::PlayerColor;
79    use crate::models::player_name::PlayerName;

82    #[tokio::test]
83    async fn save_and_load_roundtrip() {
84        let temp_dir = std::env::temp_dir();
85        let path = temp_dir.join("yatzy_test_settings.json");
86        // Clean up any previous test file
87        let _ = std::fs::remove_file(&path);
88
89        let store = SettingsStore::new(&path);
90        let settings = GameSettings::new(
91            GameMode::MultiPlayer,
92            vec![
93                PlayerSettingsEntry::new(PlayerName::new("Alice").unwrap(), PlayerColor::RED, PlayerType::Human),
94                PlayerSettingsEntry::new(PlayerName::new("Bob").unwrap(), PlayerColor::BLUE, PlayerType::Computer),
95            ],
96        );
97
98        store.save(&settings).await.unwrap();
99        let loaded = store.load().await.unwrap();
100        assert!(loaded.is_some());
101        let loaded = loaded.unwrap();
102        assert_eq!(loaded.game_mode(), GameMode::MultiPlayer);
103        assert_eq!(loaded.player_count(), 2);
104        assert_eq!(loaded.players()[0].name().as_str(), "Alice");
105
106        // Clean up
107        let _ = std::fs::remove_file(&path);
108    }
109
110    #[tokio::test]
111    async fn load_nonexistent_returns_none() {
112        let path = std::env::temp_dir().join("yatzy_nonexistent_settings.json");
113        let _ = std::fs::remove_file(&path);
114        let store = SettingsStore::new(&path);
115        let loaded = store.load().await.unwrap();
116        assert!(loaded.is_none());
117    }
