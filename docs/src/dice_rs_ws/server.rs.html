1use std::net::SocketAddr;
2use std::sync::Arc;
3
4use axum::Router;
5use axum::routing::get;
6use axum::routing::post;
7
8use crate::app_state::AppState;
9use crate::routes;
10use crate::ws_error::Result;
11use crate::ws_handler::handle_ws_upgrade;
12
13/// The dice-rs WebSocket server.
14pub struct Server {
15    router: Router,
16    bind_address: SocketAddr,
17}
18
19impl Server {
20    /// Build the axum router with all WebSocket and REST API routes.
21    pub fn build_router(state: Arc<AppState>) -> Router {
22        Router::new()
23            .route("/ws", get(handle_ws_upgrade))
24            .route("/api/scan", get(routes::scan_handler))
25            .route("/api/connect", post(routes::connect_handler))
26            .route("/api/disconnect", post(routes::disconnect_handler))
27            .route("/api/led", post(routes::led_handler))
28            .route("/api/battery", get(routes::battery_handler))
29            .route("/api/status", get(routes::status_handler))
30            .route("/api/calibrate", post(routes::calibrate_handler))
31            .with_state(state)
32    }
33
34    /// Create a new server with the given application state.
35    pub fn new(state: Arc<AppState>, bind_address: SocketAddr) -> Self {
36        Self {
37            router: Self::build_router(state),
38            bind_address,
39        }
40    }
41
42    /// Start the server.
43    pub async fn run(self) -> Result<()> {
44        let listener = tokio::net::TcpListener::bind(self.bind_address).await?;
45        tracing::info!(address = %self.bind_address, "WebSocket server listening");
46        axum::serve(listener, self.router).await?;
47        Ok(())
48    }
49}
